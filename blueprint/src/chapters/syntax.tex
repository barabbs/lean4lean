\begin{definition}[Order and equivalence of levels]
\label{def:vlevel-le-equiv}
\lean{Lean4Lean.VLevel.LE,Lean4Lean.VLevel.Equiv,Lean4Lean.VLevel.le_antisymm_iff,Lean4Lean.VLevel.equiv_def}
\leanok
\uses{def:vlevel-eval}
\srcloc{Lean4Lean/Theory/VLevel.lean}{41}
\thesisref{axioms.tex \S2.2 (order/equivalence) and \S2.5 (sort congruence)}

\lead{In short} $a\le b$ and $a\approx b$ read off \hl{pointwise comparison of
\texttt{eval}} over every assignment.
\begin{itemize}
\item $a \le b := \forall ls,\ a.\mathrm{eval}\,ls \le b.\mathrm{eval}\,ls$;
$a \approx b := \forall ls,\ a.\mathrm{eval}\,ls = b.\mathrm{eval}\,ls$.
\item Antisymmetry: $a \approx b \leftrightarrow a \le b \wedge b \le a$.
\end{itemize}
\end{definition}

\begin{lemma}[Order and congruence algebra of levels]
\label{fam:vlevel-lattice}
\lean{Lean4Lean.VLevel.le_refl,Lean4Lean.VLevel.le_trans,Lean4Lean.VLevel.zero_le,Lean4Lean.VLevel.le_succ,Lean4Lean.VLevel.succ_le_succ,Lean4Lean.VLevel.le_max_left,Lean4Lean.VLevel.le_max_right,Lean4Lean.VLevel.succ_congr,Lean4Lean.VLevel.max_congr,Lean4Lean.VLevel.imax_congr,Lean4Lean.VLevel.max_comm,Lean4Lean.VLevel.max_self,Lean4Lean.VLevel.zero_imax,Lean4Lean.VLevel.imax_zero,Lean4Lean.VLevel.imax_self,Lean4Lean.VLevel.imax_eq_zero,Lean4Lean.VLevel.IsNeverZero,Lean4Lean.VLevel.IsNeverZero.imax_eq_max}
\leanok
\uses{def:vlevel-le-equiv}
\srcloc{Lean4Lean/Theory/VLevel.lean}{45}

\lead{In short} The routine \hl{lattice algebra} of $\le$ and $\approx$.
\par\smallskip\noindent
\begin{tabular}{lp{0.82\linewidth}}
\toprule
Group & Facts \\
\midrule
Order & reflexive, transitive, $0$ least, $a\le\mathrm{succ}\,a$, two $\max$ injections \\
Congruence & $\mathrm{succ}/\max/\mathrm{imax}$ preserve $\approx$ \\
$\mathrm{imax}$ id.\ & $\mathrm{imax}(0,a)\approx a$, $\mathrm{imax}(a,0)\approx 0$,
$\mathrm{imax}(a,a)\approx a$, $\mathrm{imax}(a,b)\approx 0 \leftrightarrow b\approx 0$,
$\approx\max(a,b)$ if $b$ never zero \\
\bottomrule
\end{tabular}\par\smallskip
\end{lemma}
\begin{proof}
\uses{def:vlevel-eval}
\leanok
\lead{Idea} Each $\le$ lemma is the matching \texttt{Nat} lemma under $\forall ls$; each
$\approx$ lemma is \texttt{simp [equiv\_def, eval]} plus the matching
\texttt{Nat.max}/\texttt{imax} rewrite; \texttt{imax\_eq\_zero} adds a classical
contradiction step.
\end{proof}

\begin{definition}[Weakening \texttt{liftN} and its algebra]
\label{def:liftn}
\lean{Lean4Lean.liftVar,Lean4Lean.VExpr.liftN,Lean4Lean.VExpr.lift,Lean4Lean.VExpr.liftN_zero,Lean4Lean.VExpr.liftN'_liftN',Lean4Lean.VExpr.liftN'_liftN_lo,Lean4Lean.VExpr.liftN'_liftN_hi,Lean4Lean.VExpr.liftN_liftN,Lean4Lean.VExpr.liftN_succ,Lean4Lean.VExpr.liftN'_comm,Lean4Lean.VExpr.lift_liftN'}
\leanok
\uses{ind:vexpr}
\srcloc{Lean4Lean/Theory/VExpr.lean}{17}
\thesisref{typesys.tex, Lemma \texttt{thm:weak} (its syntactic substrate)}

\lead{In short} \texttt{liftN n e k} adds $n$ to every index of $e$ at least the cut-off
$k$, \hl{descending with $k+1$ under binders}; \texttt{lift} is \texttt{liftN 1}.
Composing two lifts needs $k_1 \le k_2 \le n_1+k_1$, specialised to low/high; lifts
at different cut-offs commute.
\end{definition}

\begin{definition}[Closed expressions]
\label{def:closedn}
\lean{Lean4Lean.VExpr.ClosedN,Lean4Lean.VExpr.Closed,Lean4Lean.VExpr.ClosedN.mono,Lean4Lean.VExpr.ClosedN.liftN_eq,Lean4Lean.VExpr.ClosedN.lift_eq,Lean4Lean.VExpr.ClosedN.liftN,Lean4Lean.VExpr.ClosedN.liftN_eq_rev}
\leanok
\uses{ind:vexpr,def:liftn}
\srcloc{Lean4Lean/Theory/VExpr.lean}{99}

\lead{In short} \texttt{e.ClosedN k} holds when every \hl{free index of $e$} is below $k$;
\texttt{Closed} is the case $k=0$.
Monotone in $k$; preserved by lifting (bound $+n$); fixed by any lift with cut-off
$\ge k$.
\end{definition}

\begin{definition}[Universe instantiation of expressions]
\label{def:instl}
\lean{Lean4Lean.VExpr.instL,Lean4Lean.VExpr.LevelWF,Lean4Lean.VExpr.instL_instL,Lean4Lean.VExpr.instL_liftN,Lean4Lean.VExpr.LevelWF.instL_id,Lean4Lean.VExpr.levelWF_liftN,Lean4Lean.VExpr.ClosedN.instL}

\srcloc{Lean4Lean/Theory/VExpr.lean}{152}

\lead{In short} \texttt{e.instL ls} substitutes \texttt{ls} into every sort and every

\begin{definition}[Single-variable instantiation and the lift/instantiation square]
\label{def:inst}
\lean{Lean4Lean.VExpr.instVar,Lean4Lean.VExpr.inst,Lean4Lean.VExpr.liftN_instN_lo,Lean4Lean.VExpr.liftN_instN_hi,Lean4Lean.VExpr.liftN_inst_hi,Lean4Lean.VExpr.lift_instN_lo,Lean4Lean.VExpr.lift_inst_hi,Lean4Lean.VExpr.inst_liftN,Lean4Lean.VExpr.inst_lift,Lean4Lean.VExpr.inst_liftN_lo,Lean4Lean.VExpr.inst_liftN_bvar}
\leanok
\uses{def:liftn,ind:vexpr}
\srcloc{Lean4Lean/Theory/VExpr.lean}{194}
\thesisref{typesys.tex, Lemma \texttt{thm:subst} (its syntactic substrate)}

\lead{In short} \texttt{e.inst a k} substitutes $a$ for variable $k$, \hl{shifting
variables above $k$ down by one} and lifting $a$ by $k$; \texttt{instVar} is the
three-way test $i<k$, $i=k$, $i>k$.
The lifting/instantiation square: \texttt{lo} (instantiation above the lift),
\texttt{hi} (below it), plus the cancellation
$(\texttt{liftN}\,1\,e\,k).\texttt{inst}\,a\,k = e$.
\end{definition}

\begin{definition}[Telescope instantiation]
\label{def:insts}
\lean{Lean4Lean.VExpr.insts,Lean4Lean.VExpr.insts_append,Lean4Lean.VExpr.insts_sort,Lean4Lean.VExpr.insts_const,Lean4Lean.VExpr.insts_app,Lean4Lean.VExpr.insts_nil,Lean4Lean.VExpr.insts_cons}
\leanok
\uses{def:inst}
\srcloc{Lean4Lean/Theory/VExpr.lean}{333}
\thesisref{axioms.tex \S2.6.1, the telescope substitution}

\lead{In short} \texttt{e.insts as} substitutes a whole telescope, one \texttt{inst} at
index $0$ per element, \hl{the head of the list substituted last}, so
\texttt{insts (as ++ bs)} is \texttt{insts as} after \texttt{insts bs}.
\lead{Why} A substitution, not an applied abstraction: substituted pieces decompose
on the nose, since substitution (unlike application) commutes syntactically.
\end{definition}

\begin{definition}[Inverting a lift: \texttt{unliftN} and \texttt{Skips}]
\label{def:unliftn-skips}
\lean{Lean4Lean.VExpr.unliftN,Lean4Lean.VExpr.Skips,Lean4Lean.VExpr.Skips',Lean4Lean.VExpr.skips_iff,Lean4Lean.VExpr.skips_iff_exists,Lean4Lean.VExpr.skips_add,Lean4Lean.VExpr.liftN_inj,Lean4Lean.VExpr.of_liftN_eq_liftN,Lean4Lean.VExpr.of_liftN_eq_instL,Lean4Lean.VExpr.Skips.of_liftN_hi,Lean4Lean.VExpr.Skips.of_instL}
\leanok
\uses{def:liftn,def:inst}
\srcloc{Lean4Lean/Theory/VExpr.lean}{367}

\lead{In short} \texttt{unliftN e n k} undoes $n$ lifts at cut-off $k$ by instantiating
\texttt{default} $n$ times; \texttt{e.Skips n k} says \hl{$e$ is in the image of
\texttt{liftN n \_ k}} ($\texttt{liftN}\,n\,(\texttt{unliftN}\,e\,n\,k)\,k = e$), and
\texttt{Skips'} restates it structurally.
Consequences: \texttt{liftN} is injective; equal lifts share a preimage; skipping
is inherited from a larger lift or \texttt{instL}.
\end{definition}

\section{The \texttt{Lift} and \texttt{Subst} algebras}

\begin{definition}[The \texttt{Lift} renaming algebra]
\label{ind:lift}
\lean{Lean4Lean.Lift,Lean4Lean.Lift.skipN,Lean4Lean.Lift.consN,Lean4Lean.Lift.comp,Lean4Lean.Lift.dom,Lean4Lean.Lift.size,Lean4Lean.Lift.depth,Lean4Lean.Lift.inter,Lean4Lean.Lift.diff,Lean4Lean.Lift.Fixes,Lean4Lean.Lift.liftVar,Lean4Lean.Lift.liftVar_comp,Lean4Lean.Lift.Fixes.liftVar_eq,Lean4Lean.Lift.depth_succ}
\leanok
\uses{ind:vexpr}
\srcloc{Lean4Lean/Theory/VExpr.lean}{587}

\lead{In short} A finite description of an \hl{order-preserving injection of de Bruijn
indices}: \texttt{refl}, \texttt{skip} (insert a binder), \texttt{cons} (go under
one).
\begin{itemize}
\item Acts on variables via \texttt{liftVar}; carries a composition monoid
(contravariant), a meet/difference, the measures \texttt{dom}/\texttt{size}/\texttt{depth},
and \texttt{Fixes k} (identity below $k$).
\item Generalises $\texttt{liftN}\,n\,\_\,k$, recovered as
$\texttt{consN}\,(\texttt{skipN}\,\texttt{refl}\,n)\,k$, to arbitrary interleavings.
\end{itemize}
\end{definition}

\begin{lemma}[\texttt{consN} fixes the binders it is pushed under]
\label{fam:lift-consn}
\lean{Lean4Lean.Lift.consN_fixes,Lean4Lean.Lift.consN_cons,Lean4Lean.Lift.liftVar_consN_lt,Lean4Lean.Lift.liftVar_consN_succ}
\leanok
\contrib{trproj} \srcloc{Lean4Lean/Theory/VExpr.lean}{744}
\uses{ind:lift}
\thesisref{none}

\lead{In short} Four facts about \hl{pushing a lift under $m$ binders}.
$(\rho.\texttt{consN}\,m)$ fixes $m$; $(\texttt{cons}\,\rho).\texttt{consN}\,j =
\rho.\texttt{consN}\,(j{+}1)$; the renaming is the identity below $m$; it commutes with
successor one binder at a time.
\end{lemma}
\begin{proof}
\uses{ind:lift}
\leanok
\lead{Idea} \texttt{consN\_fixes} and \texttt{consN\_cons} recurse structurally on $m$/$j$;
\texttt{liftVar\_consN\_lt} feeds \texttt{consN\_fixes} to \texttt{Lift.Fixes.liftVar\_eq};
\texttt{liftVar\_consN\_succ} is \texttt{rfl}.
\end{proof}

\begin{definition}[General (simultaneous) substitutions]
\label{def:subst}
\lean{Lean4Lean.VExpr.Subst,Lean4Lean.VExpr.subst,Lean4Lean.VExpr.Subst.lift,Lean4Lean.VExpr.Subst.liftN,Lean4Lean.VExpr.Subst.lift_l,Lean4Lean.VExpr.Subst.lift_r,Lean4Lean.VExpr.subst_lift',Lean4Lean.VExpr.lift'_subst,Lean4Lean.VExpr.Subst.comp,Lean4Lean.VExpr.Subst.comp_lift,Lean4Lean.VExpr.subst_subst,Lean4Lean.VExpr.liftN_subst}
\leanok
\uses{def:lift-prime,ind:vexpr}
\srcloc{Lean4Lean/Theory/VExpr.lean}{794}

\lead{In short} A \texttt{Subst} is a function \texttt{Nat} $\to$ \texttt{VExpr};
\texttt{e.subst sigma} applies it, \hl{lifting under binders} (\texttt{Subst.lift},
iterated as \texttt{liftN}).
\begin{itemize}
\item \texttt{Subst.comp} composes two substitutions, with action law
$(e.\texttt{subst}\,\sigma).\texttt{subst}\,\sigma' = e.\texttt{subst}\,(\sigma.\texttt{comp}\,\sigma')$;
only \texttt{comp\_lift} is stated of associativity/unitality.
\item A \texttt{Lift} absorbs into a substitution from either side, rewriting into
\texttt{Subst.lift\_l}/\texttt{lift\_r} --- the API \ref{thm:lift-eq-subst} onward builds
on.
\end{itemize}
\end{definition}

\section{Weakening and instantiation as substitutions}

Weakening and instantiation are both substitutions (\ref{thm:lift-eq-subst}): a builder
lemma need only be proved once, against \texttt{subst}.

\begin{theorem}[Weakening is a substitution]
\label{thm:lift-eq-subst}
\lean{Lean4Lean.VExpr.lift'_eq_subst}
\leanok
\contrib{trproj} \srcloc{Lean4Lean/Theory/VExpr.lean}{970}
\uses{def:subst,def:lift-prime}
\thesisref{none}

\lead{In short} \hl{Weakening along $\rho$ is a substitution}:
$\texttt{e.lift'}\,\rho = \texttt{e.subst}\,(\texttt{Subst.lift\_l}\,\rho\,\texttt{Subst.id})$,
renaming each variable along $\rho$.
\end{theorem}
\begin{proof}
\uses{fam:subst-basic,def:subst}
\leanok
\lead{Idea} \texttt{subst\_id.symm.trans subst\_lift'}: run the identity substitution on
$\texttt{e.lift'}\,\rho$, then absorb the lift on the left. The section's keystone.
\end{proof}

\begin{lemma}[\texttt{Subst.liftN} bookkeeping]
\label{fam:subst-liftn-algebra}
\lean{Lean4Lean.VExpr.Subst.lift_liftN,Lean4Lean.VExpr.Subst.liftN_liftN,Lean4Lean.VExpr.Subst.lift_lift_l_id,Lean4Lean.VExpr.Subst.liftN_lift_l_id,Lean4Lean.VExpr.Subst.Fixes.liftN}
\leanok
\contrib{trproj} \srcloc{Lean4Lean/Theory/VExpr.lean}{973}
\uses{def:subst,def:lift-prime}
\thesisref{none}

\lead{In short} Five unfolding lemmas for \hl{iterated lifting} of a substitution.
\begin{itemize}
\item $\sigma.\texttt{lift}.\texttt{liftN}\,n = \sigma.\texttt{liftN}\,(n{+}1)$;
$(\sigma.\texttt{liftN}\,k).\texttt{liftN}\,j = \sigma.\texttt{liftN}\,(k{+}j)$.
\item $\texttt{Subst.lift\_l}\,\rho\,\texttt{id}$ commutes with \texttt{lift}/\texttt{liftN},
turning $\rho$ into $\rho.\texttt{cons}$/$\rho.\texttt{consN}\,j$; and
$\sigma.\texttt{liftN}\,n$ fixes $n$.
\item Makes \texttt{Lift.consN} line up with \texttt{Subst.liftN} binder for binder, the
section's design goal.
\end{itemize}
\end{lemma}
\begin{proof}
\uses{fam:subst-basic}
\leanok
\lead{Idea} \texttt{Subst.lift\_liftN}/\texttt{liftN\_liftN} recurse on the lift count via
\texttt{congrArg Subst.lift}; \texttt{Fixes.liftN} recurses through \texttt{Fixes.lift};
\texttt{liftN\_lift\_l\_id} rewrites by \texttt{lift\_lift\_l\_id}, itself
\texttt{lift\_l\_lift} plus \texttt{id\_lift}.
\end{proof}

\begin{lemma}[Substitution past a one-variable instantiation, at the level of substitutions]

\end{proof}

\section{Level equivalence is coNP-hard}

No thesis counterpart: a reduction from CNF satisfiability to \texttt{VLevel}
equivalence, a lower bound for \texttt{Level.isEquiv}/\texttt{Level.geq}.

\begin{definition}[CNF formulas and the dual-rail encoding]
\label{def:levelsat-cnf}
\lean{Lean4Lean.LevelSat.Lit,Lean4Lean.LevelSat.Clause,Lean4Lean.LevelSat.CNF,Lean4Lean.LevelSat.ClauseSat,Lean4Lean.LevelSat.Sat,Lean4Lean.LevelSat.yes,Lean4Lean.LevelSat.no,Lean4Lean.LevelSat.rail}
\leanok
\srcloc{Lean4Lean/Theory/LevelSat.lean}{70}

\lead{In short} Literals: pairs in $\mathrm{Fin}\,n \times \mathrm{Bool}$; clauses:
lists of literals; a CNF: a list of clauses; \texttt{Sat}: the existential over Boolean
assignments.
\begin{itemize}
\item Each variable $i$ gets two rails: $i^{+}$ at index $2i$, $i^{-}$ at $2i+1$, \hl{nonzero
meaning true}.
\item Indexing variables by $\mathrm{Fin}\,n$ builds ``only variables with rails'' into
the type, avoiding a side condition.
\end{itemize}
\end{definition}

\begin{definition}[The two levels of the reduction]
\label{def:levelsat-levels}
\lean{Lean4Lean.LevelSat.two,Lean4Lean.LevelSat.clauseLvl,Lean4Lean.LevelSat.gates,Lean4Lean.LevelSat.var,Lean4Lean.LevelSat.small,Lean4Lean.LevelSat.big}
\leanok
\uses{def:levelsat-cnf,ind:vlevel}
\srcloc{Lean4Lean/Theory/LevelSat.lean}{104}

\lead{In short} $\mathrm{clauseLvl}\,C$ is the $\max$ of a clause's rails;
$\mathrm{gates}\,\varphi$ gates the constant $2$ behind every clause by an iterated
$\mathrm{imax}$ fold.
\begin{itemize}
\item $\mathrm{var}\,i = \max(i^{+},\,\mathrm{imax}(\mathrm{imax}(2,i^{+}),\,i^{-}))$
detects a variable with \hl{both rails nonzero}; $\mathrm{small}\,n = \max_{i<n}\mathrm{var}\,i$;
$\mathrm{big}\,\varphi = \max(\mathrm{small}\,n,\,\mathrm{gates}\,\varphi)$.
\item Every level is monotone, so negation enters as a failure of domination, not a
computed value.
\end{itemize}
\end{definition}

\begin{lemma}[Evaluation lemmas for the gadget levels]
\label{fam:levelsat-eval}
\lean{Lean4Lean.LevelSat.eval_var_eq,Lean4Lean.LevelSat.yes_le_eval_var,Lean4Lean.LevelSat.no_le_eval_var,Lean4Lean.LevelSat.two_le_eval_var,Lean4Lean.LevelSat.eval_var_le_one,Lean4Lean.LevelSat.le_eval_small,Lean4Lean.LevelSat.two_le_eval_small,Lean4Lean.LevelSat.eval_small_le_one,Lean4Lean.LevelSat.eval_clauseLvl_ne_zero,Lean4Lean.LevelSat.eval_foldl_imax_le,Lean4Lean.LevelSat.eval_foldl_imax_le_of_zero,Lean4Lean.LevelSat.le_eval_foldl_imax}
\leanok
\uses{def:levelsat-levels,def:vlevel-eval}
\srcloc{Lean4Lean/Theory/LevelSat.lean}{141}

\lead{In short} Pointwise \hl{evaluation of the gadgets}.
\par\smallskip\noindent
\begin{tabular}{lp{0.83\linewidth}}
\toprule
Group & Facts \\
\midrule
$\mathrm{var}\,i$ & the larger rail, or $\ge 2$ if both are nonzero \\
$\mathrm{small}\,n$ & dominates parameters below $2n$; $\ge 2$ if some $i<n$ has both
rails nonzero, else $\le 1$ \\
$\mathrm{imax}$ fold & bounded by the gates, unless all nonzero, when the gated constant
survives \\
\bottomrule
\end{tabular}\par\smallskip
\end{lemma}
\begin{proof}
\uses{def:vlevel-eval}
\leanok
\lead{Idea} Each lemma restates \texttt{VLevel.eval} via \texttt{max}/\texttt{ite} and
closes with \texttt{split}/\texttt{omega}; the \texttt{small} lemmas induct on $n$, the
fold lemmas on the gate list.
\end{proof}

\begin{definition}[The rail encoding of an assignment]
\label{def:levelsat-raillist}
\lean{Lean4Lean.LevelSat.railVal,Lean4Lean.LevelSat.railList,Lean4Lean.LevelSat.ofFin,Lean4Lean.LevelSat.getD_railList,Lean4Lean.LevelSat.getD_railList_rail}
\leanok
\uses{def:levelsat-cnf}
\srcloc{Lean4Lean/Theory/LevelSat.lean}{270}

\lead{In short} $\mathrm{railVal}\,a\,j$ is $1$ when rail $j$ agrees with assignment $a$,
else $0$; $\mathrm{railList}\,n\,a$ lists the \hl{first $2n$ rail values}, the point where
$\mathrm{big}$ and $\mathrm{small}$ differ when $a$ satisfies the formula.
\texttt{ofFin} reads a $\mathrm{Fin}\,n$-indexed assignment at an arbitrary index.
\end{definition}

\begin{theorem}[Level equivalence is coNP-hard]
\label{thm:levelsat-equiv-iff-unsat}
\lean{Lean4Lean.LevelSat.equiv_iff_unsat}
\leanok
\uses{def:levelsat-levels,def:levelsat-cnf,def:vlevel-le-equiv}
\srcloc{Lean4Lean/Theory/LevelSat.lean}{331}

\lead{In short} For every CNF $\varphi$ over $n$ variables, $\mathrm{big}\,\varphi \approx \mathrm{small}\,n$
iff $\varphi$ is \hl{unsatisfiable}.
Both levels are linear in $\varphi$ (\ref{fam:levelsat-size}), so deciding
\texttt{VLevel.Equiv} --- what \texttt{Level.isEquiv} solves --- is coNP-hard.
\end{theorem}
\begin{proof}
\uses{fam:levelsat-eval,def:levelsat-raillist}
\leanok
\lead{Idea} Forward: a satisfying assignment gives a point where $\mathrm{small}\le 1$
while every gate is nonzero, so the gated $2$ exceeds it. Backward: a differing point
forces every gate nonzero and consistent rails, giving a satisfying assignment.
\end{proof}

\begin{lemma}[The same hardness for the order test]
\label{thm:levelsat-le-iff-unsat}
\lean{Lean4Lean.LevelSat.le_iff_unsat,Lean4Lean.LevelSat.small_le_big}
\leanok
\uses{def:levelsat-levels,def:vlevel-le-equiv}
\srcloc{Lean4Lean/Theory/LevelSat.lean}{382}

\lead{In short} $\mathrm{big}\,\varphi \le \mathrm{small}\,n$ iff $\varphi$ is
unsatisfiable, so \hl{\texttt{Level.geq} is coNP-hard} too.
\end{lemma}
\begin{proof}
\uses{thm:levelsat-equiv-iff-unsat}
\leanok
\lead{Idea} Rewrite by \ref{thm:levelsat-equiv-iff-unsat} and antisymmetry; the missing
half, $\mathrm{small}\,n \le \mathrm{big}\,\varphi$, holds structurally by
\texttt{Nat.le\_max\_left}.
\end{proof}

\begin{lemma}[The reduction is linear]
\label{fam:levelsat-size}
\lean{Lean4Lean.LevelSat.size,Lean4Lean.LevelSat.cnfSize,Lean4Lean.LevelSat.size_clauseLvl,Lean4Lean.LevelSat.size_gates,Lean4Lean.LevelSat.size_var,Lean4Lean.LevelSat.size_small,Lean4Lean.LevelSat.size_big}
\leanok
\uses{def:levelsat-levels,def:levelsat-cnf}
\srcloc{Lean4Lean/Theory/LevelSat.lean}{392}

\lead{In short} Counting nodes,
$\mathrm{size}(\mathrm{big}\,\varphi) = 10n + 2\,\mathrm{cnfSize}(\varphi) + 5$, so both
levels are \hl{linear in the formula} --- what makes
\ref{thm:levelsat-equiv-iff-unsat} a hardness result.
\lead{Caveat} Only the size half of the linear-time claim is formalised.
\end{lemma}
\begin{proof}
\uses{def:levelsat-levels}
\leanok
\lead{Idea} Inductions on the clause, the clause list and $n$, each closing with
\texttt{omega}; $\mathrm{size}(\mathrm{var}\,i)=9$ and $\mathrm{size}(2)=3$ are
\texttt{rfl}.
\end{proof}

\begin{example}[Two worked instances]
\label{ex:levelsat-instances}
\lean{Lean4Lean.LevelSat.selfContradiction,Lean4Lean.LevelSat.not_sat_selfContradiction}
\leanok
\uses{thm:levelsat-equiv-iff-unsat}
\srcloc{Lean4Lean/Theory/LevelSat.lean}{431}

\lead{In short} $x_0 \wedge \neg x_0$ is unsatisfiable, so its two levels are equivalent;
$x_0$ alone is satisfiable, and the levels \hl{differ at $[1,0]$} ($\mathrm{big}=2$,
$\mathrm{small}=1$).
The first case is where a naive reduction, reading each parameter as a SAT
variable, would misfire: both rails of $x_0$ could be nonzero; $\mathrm{small}$ rules
that out.
\end{example}

\section*{Caveats}

\begin{itemize}
\item \texttt{VRecursor.k} is recorded but unused: the thesis's second $\iota$ rule
(K-like reduction) is not modelled.
\item The quotient computation rule lives in \texttt{defeqs}, not \texttt{pats}: two
mechanisms for computation rules coexist.
\item \texttt{VRecRule.ctorParams} is redundant for direct inductive blocks; needed only
once nested blocks are admitted.
\item Universe order/equivalence are defined semantically, over all assignments, not by
the thesis's syntactic derivation system.
\item Only coNP-hardness is formalised for \texttt{Level.isEquiv}/\texttt{Level.geq}; the
matching NP upper bound is sketched, not proved.
\item The quotient constant signatures come from the ambient Lean version, not from a
written-out model.
\end{itemize}
