\ref{thm:tr-env-proj-defeq}, \ref{thm:tr-expr-mk-app-list}).

\item \lead{Lean files} \texttt{Lean4Lean/Verify/Environment/\{Basic,Lemmas,Quot,

\section{The inductive-block witness}

\begin{definition}[Type-former translation, \texttt{TrIndType}]
\label{struct:tr-ind-type}
\contrib{mixed}
\lean{Lean4Lean.TrIndType}
\leanok
\uses{def:tr-constant, struct:vinductivetype, def:pi-telescope}
\srcloc{Lean4Lean/Verify/Environment/Basic.lean}{216}
\thesisref{axioms.tex, \S\S ``Inductive types'' and ``The recursor''.}

\lead{In short} A kernel type former \hl{translates together with its constructors}.
\begin{itemize}
\item $ival$ translates to the model \texttt{VInductiveType} $t$ in $env_1$;
$ival.\mathtt{ctors}$ names $cvals$ exactly.
\item Each \texttt{ctorInfo} of $cvals$ translates in $env_T$, with
$cval.\mathtt{numParams} + cval.\mathtt{numFields} = c.\mathtt{type}.\mathtt{piArity}$.
\item $ival.\mathtt{all} = block$; $ival.\mathtt{numParams} = nparams$;
$ival.\mathtt{numIndices} = t.\mathtt{type}.\mathtt{piArity} - nparams$.
\end{itemize}
\lead{Caveat} Three environments: type formers translate in $env_1$, constructors in
$env_T$, recursor reducts (Definition~\ref{struct:tr-recursor}) in $env_R$.
\end{definition}

\begin{definition}[Recursor translation, \texttt{TrRecursor}]
\label{struct:tr-recursor}
\contrib{iota}
\lean{Lean4Lean.TrRecursor}
\leanok
\uses{def:tr-constant, struct:vrecursor, ind:trexprs}
\srcloc{Lean4Lean/Verify/Environment/Basic.lean}{237}
\thesisref{axioms.tex, \S ``The computation rule ($\iota$ reduction)''.}

\lead{In short} A kernel \texttt{recInfo} \hl{translates rule for rule} to a model
\texttt{VRecursor}.
\begin{itemize}
\item $rval$ translates to $r$ in $env_C$; the telescope split, the \texttt{all} list and
the K-like flag \texttt{k} are copied from $rval$.
\item Rules match one-to-one: same constructor and field count, parameter count read off
$m_2$, reduct translating in $env_R$.
\item \texttt{name\_major} (\texttt{trproj}): the kernel names a recursor
$\mathtt{mkRecName}\ n$ after the type former $n$ it eliminates.
\end{itemize}
\lead{Caveat} \texttt{all} \srcloc{Lean4Lean/Verify/Environment/Basic.lean}{240} and
\texttt{k} \srcloc{Lean4Lean/Verify/Environment/Basic.lean}{245} are carried, but no
model rule reads \texttt{k}.
\end{definition}

\label{struct:add-induct}

\begin{theorem}[Kernel recursor lookup vs. the block]
\label{thm:add-induct-rec-find}
\contrib{iota}
\lean{Lean4Lean.AddInduct.rec_find}
\leanok
\uses{struct:add-induct, struct:tr-recursor}
\srcloc{Lean4Lean/Verify/Environment/Basic.lean}{434}
\thesisref{axioms.tex, \S ``The computation rule ($\iota$ reduction)''.}

\lead{In short} If $m_1$ well-formed and $m_2$ resolves $recName$ to kernel $rval$, then

$r$ -- same constructor and field count, parameter count as in $m_2$, closed reduct

\end{theorem}
\begin{proof}
\uses{family:add-induct-find, family:indlem-closed, family:add-induct-stages}
\leanok
\lead{Idea} \texttt{AddInduct.find?} splits; the block case gives the \texttt{TrRecursor},
closedness and translation following from \texttt{addRules\_closed} and \texttt{leR}.
\end{proof}

\begin{theorem}[Model recursor is registered]
\label{thm:add-induct-rec-reg}
\contrib{iota}
\lean{Lean4Lean.AddInduct.rec_reg}
\leanok
\uses{struct:add-induct, struct:tr-recursor}
\srcloc{Lean4Lean/Verify/Environment/Basic.lean}{464}
\thesisref{axioms.tex, \S ``The computation rule ($\iota$ reduction)''.}

\lead{In short} Given $m_1$ well-formed and $decl.\mathtt{WF}\ env_1$: for each
$r \in decl.\mathtt{recs}$ there is a kernel $rval$ with
$m_2.\mathtt{find?}\ r.\mathtt{name} = \mathtt{some}\ (\mathtt{recInfo}\ rval)$, same major
index and telescope split, and every model rule of $r$ found (by constructor) among
$rval$'s, with matching field count, parameter count in $m_2$, and reduct: \hl{every model
recursor is registered} under its kernel name.
\end{theorem}
\begin{proof}
\uses{family:add-induct-find, structure:ind-wf, family:add-induct-stages}
\leanok
\lead{Idea} \texttt{find?\_self} places the recursor in $m_2$; distinctness of kernel
rules' constructors (\texttt{rules\_nodup}) turns membership into a \texttt{find?} result.
\end{proof}

\begin{theorem}[Constructor lookup yields its $\iota$ rule]
\label{thm:add-induct-ctor-find}
\contrib{iota}
\lean{Lean4Lean.AddInduct.ctor_find}
\leanok
\uses{struct:add-induct, struct:tr-ind-type}

\thesisref{axioms.tex, \S ``The computation rule ($\iota$ reduction)''.}

\lead{In short} Given $m_1$ well-formed, $decl.\mathtt{WF}\ env_1$, and $m_2$ resolving
$ctorName$ to kernel $cval$: either $m_1$ already did, or there are
$t \in decl.\mathtt{types}$, $c \in t.\mathtt{ctors}$ named $ctorName$, and a recursor
$r \in decl.\mathtt{recs}$ with a rule $ru$ firing on $ctorName$ such that
$ru.\mathtt{ctorParams} = cval.\mathtt{numParams}$,
$ru.\mathtt{nfields} = cval.\mathtt{numFields}$, $r.\mathtt{numParams} =

\end{theorem}
\begin{proof}
\uses{family:add-induct-find, family:ind-wf-corollaries, thm:indlem-rules-ctor-shape}
\leanok
\lead{Idea} Splits via \texttt{find?}; \texttt{TrIndType} gives the rule
(\texttt{ctors\_have\_rules}), and its arity conjunct forces the field count by
\texttt{omega}.
\end{proof}

\begin{lemma}[Basic consequences of alignment]
\label{family:aligned-basic}
\contrib{mixed}
\lean{Lean4Lean.Aligned.map_wf, Lean4Lean.Aligned.find?_iff, Lean4Lean.Aligned.find?, Lean4Lean.Aligned.find?_uniq}
\leanok
\srcloc{Lean4Lean/Verify/Environment/Lemmas.lean}{35}

\lead{In short} The constant map is well-formed (\texttt{map\_wf}); a name resolves in $C$
visible at $safety$ iff in $venv.\mathtt{constants}$ (\texttt{find?\_iff}), producing a
model constant translating the kernel one (\texttt{find?}); a kernel and model lookup of

\section*{Caveats}
\begin{itemize}
\item The \texttt{inductDecl} case of \ref{thm:add-decl-wf} is \texttt{sorry}, and nothing
builds an \texttt{AddInduct} witness from \texttt{Environment.addInductive}: the whole
inductive-block apparatus is conditional on that hypothesis.
\item \ref{thm:tr-env-proj-defeq} has no non-test consumer: \texttt{reduceProjCore.WF},
\texttt{inferProj.WF\_struct} and \texttt{inferProj.WF} are still \texttt{sorry}, and its
premise \texttt{TrProjCtor} is inhabited only by hand in tests.
\item \ref{family:beta-telescope} and \ref{thm:tr-env-proj-defeq} inherit
\texttt{sorryAx} through unique typing (\texttt{VEnv.WF.orderedStrong},
\texttt{VEnv.WF.patsStrong}).
\item The \ref{family:persistent-hash-map-axioms} axioms sit on \ref{thm:tr-env-proj-defeq}'s
critical path, alongside \texttt{propext}, \texttt{Classical.choice}, \texttt{Quot.sound}.
\item The \ref{family:expr-axioms} axioms are the project's largest trust assumption:
every kernel-level \texttt{Expr} theorem rests on them, checked only by inspection.
\end{itemize}
