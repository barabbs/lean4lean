\chapter{Soundness of the type checker}
\label{chap:typechecker}

\section*{At a glance}
\begin{itemize}
\item \lead{Goal} Partial correctness of the executable checker against the model: each kernel
  function \texttt{f} gets a companion \texttt{f.WF} making \hl{whatever \texttt{f} returns}
  justified, if the inputs translate. Failure and non-termination are unconstrained: soundness
  only.
\item \lead{Main results} Theorem~\ref{thm:vtc-whnf}, \ref{thm:vtc-isdefeqcore},
  \ref{thm:vtc-infertype-prime}, \ref{thm:vtc-withfuel}, Lemma~\ref{thm:vtc-toplevel}.
\item \lead{Status} \stProved{} program logic and binder abstraction; \stTainted{} WHNF, type
  inference, definitional equality, top level; \stSorry{} six lemmas:
  Lemma~\ref{thm:vtc-reduceprojcore}, Theorem~\ref{thm:vtc-reducerecursor},
  Lemma~\ref{thm:vtc-tryetastruct}, Lemma~\ref{thm:vtc-isdefequnitlike},
  Theorem~\ref{thm:vtc-inferproj-struct}, Theorem~\ref{thm:vtc-inferproj}.
\item \lead{Contribution} \texttt{iota}: the pure $\iota$ step, proved
  (\ref{thm:vtc-iota-reduce-core}), unwired to \ref{thm:vtc-reducerecursor}; and the
  ambient-context generalization of the checker's soundness (\ref{thm:vtc-m-run-ambient}).
  \texttt{trproj}: repairs \ref{thm:vtc-inferproj}, adds scoped \ref{thm:vtc-inferproj-struct};
  both open.
\item \lead{Lean files} \texttt{Verify/TypeChecker/\{Basic,Reduce,WHNF,InferType,IsDefEq\}.lean},
  \texttt{Verify/TypeChecker.lean}.
\item \lead{Thesis} Indirect: $\Leftrightarrow$ is never formalised, so
  Theorem~\ref{thm:vtc-isdefeqcore} stands in for it; the state invariant
  (Definition~\ref{def:vtc-vstate-wf}) has no thesis counterpart at all.
\item \lead{Where} $\Delta$: model local context; $\Gamma$: its list of types. Primed ($e'$):
  model terms; unprimed: kernel \texttt{Expr}. ``translates'' abbreviates \texttt{TrExprS}; its
  up-to-defeq variant is \texttt{TrExpr}.
\end{itemize}

\section{The verification context}

\begin{definition}[Model local context, \texttt{MLCtx}]
  \label{def:vtc-mlctx}
  \lean{Lean4Lean.TypeChecker.MLCtx, Lean4Lean.TypeChecker.MLCtx.vlctx, Lean4Lean.TypeChecker.MLCtx.lctx, Lean4Lean.TypeChecker.MLCtx.length, Lean4Lean.TypeChecker.MLCtx.decls, Lean4Lean.TypeChecker.MLCtx.fvarRevList, Lean4Lean.TypeChecker.MLCtx.letValList, Lean4Lean.TypeChecker.MLCtx.dropN}
  \leanok
  \uses{def:vt-vlctx}
  \srcloc{Lean4Lean/Verify/TypeChecker/Basic.lean}{106}

  \lead{In short} A snoc-list keeping the kernel \texttt{LocalContext} and $\Delta$ \hl{in
  lockstep}: binders \texttt{vlam}/\texttt{vlet} carry kernel data (id, name, type, binder info,
  \texttt{let} value) and model data (\texttt{VExpr} type, value); \texttt{vlctx}/\texttt{lctx}
  project to each; \texttt{dropN} etc.\ expose the last $n$ binders.
\end{definition}

\begin{definition}[Well-formedness of an \texttt{MLCtx}]
  \label{def:vtc-mlctx-wf}
  \lean{Lean4Lean.TypeChecker.MLCtx.WF, Lean4Lean.TypeChecker.MLCtx.WF.tr, Lean4Lean.TypeChecker.MLCtx.WF.dropN, Lean4Lean.TypeChecker.MLCtx.dropN_fvars_subset, Lean4Lean.TypeChecker.MLCtx.noBV}

  \srcloc{Lean4Lean/Verify/TypeChecker/Basic.lean}{158}
  \thesisref{typesys.tex, Regularity}

  \lead{In short} Recursion on \texttt{MLCtx}: each binder's fvar is fresh below it, its type
  translates to a model type (\texttt{vlet}: also its value); \hl{\texttt{WF.tr}} packages this
  as \texttt{TrLCtx c.lctx c.vlctx}, and \texttt{noBV} records no bound-variable entries.
\end{definition}

  \label{def:vtc-vcontext}
  \lean{Lean4Lean.TypeChecker.VContext, Lean4Lean.TypeChecker.VContext.trlctx, Lean4Lean.TypeChecker.VContext.Ewf, Lean4Lean.TypeChecker.VContext.Δwf, Lean4Lean.TypeChecker.VContext.TrExprS, Lean4Lean.TypeChecker.VContext.TrExpr, Lean4Lean.TypeChecker.VContext.TrTyping, Lean4Lean.TypeChecker.VContext.FVarsBelow, Lean4Lean.TypeChecker.VContext.MLCWF}

  \uses{def:vtc-mlctx-wf, ind:tr-env, ind:primspec, def:isfvarupset}

  runtime one; \texttt{nonrec abbrev}s specialise the model relations here.
\end{definition}

\begin{definition}[Checker state invariant, \texttt{VState.WF}]
  \label{def:vtc-vstate-wf}
  \lean{Lean4Lean.TypeChecker.VState, Lean4Lean.InferCache.WF, Lean4Lean.TypeChecker.WHNFCache.WF, Lean4Lean.TypeChecker.UnfoldCache.WF, Lean4Lean.TypeChecker.VState.WF, Lean4Lean.TypeChecker.VState.LE, Lean4Lean.TypeChecker.VState.WF.find?_eq_none}

  \lead{In short} What every \hl{mutable field} of the state must satisfy: the local context

\section{The program logic}

\begin{definition}[Partial-correctness predicate for the checker monad, \texttt{M.WF}]
  \label{def:vtc-m-wf}
  \lean{Lean4Lean.TypeChecker.M.WF, Lean4Lean.TypeChecker.M.WF.bind, Lean4Lean.TypeChecker.M.WF.pure, Lean4Lean.TypeChecker.M.WF.map, Lean4Lean.TypeChecker.M.WF.mono, Lean4Lean.TypeChecker.M.WF.throw, Lean4Lean.TypeChecker.M.WF.le}
  \leanok
  \uses{def:vtc-vstate-wf}
  \srcloc{Lean4Lean/Verify/TypeChecker/Basic.lean}{278}
  \thesisref{typesys.tex, \texttt{item:alg\_defn}}

  \lead{In short} \texttt{M.WF c s x Q}: if \texttt{s} satisfies the invariant and \texttt{x} run
  at \texttt{c}, \texttt{s} returns \texttt{.ok (a, s')}, some \texttt{VState} for \texttt{s'}
  extends \texttt{s}, keeps the invariant, and \hl{\texttt{Q a} holds}. A throw or non-terminating
  run holds vacuously: soundness only, throughout. Combinators \texttt{bind}/\texttt{pure}/
  \texttt{map}/\texttt{mono}/\texttt{throw}/\texttt{le} unfold the monad stack.
\end{definition}

\begin{definition}[Soundness contract of the method table, \texttt{Methods.WF}]
  \label{def:vtc-methods-wf}
  \lean{Lean4Lean.TypeChecker.Methods.WF}
  \leanok
  \uses{def:vtc-m-wf, def:isfvarupset}
  \srcloc{Lean4Lean/Verify/TypeChecker/Basic.lean}{314}
  \thesisref{axioms.tex, \S Definitional equality and head reduction}

  \lead{In short} What the kernel's four mutually recursive entry points must satisfy.
  \begin{itemize}
  \item \texttt{isDefEqCore} true on translating terms yields \texttt{IsDefEqU};
  \item \texttt{whnfCore}, \texttt{whnf} return a term \texttt{FVarsBelow} the input translating
    like it;
  \item \texttt{inferType}, given fvars in scope (at \texttt{inferOnly}, already translating),
    returns \texttt{ty} with $\exists\, e'\, ty'$, \texttt{TrTyping e ty e' ty'}.
  \end{itemize}
\end{definition}

  \label{def:vtc-recm-wf}
  \lean{Lean4Lean.TypeChecker.RecM.WF, Lean4Lean.TypeChecker.M.WF.liftExcept, Lean4Lean.TypeChecker.M.WF.lift, Lean4Lean.TypeChecker.RecM.WF.bind, Lean4Lean.TypeChecker.RecM.WF.bind_le, Lean4Lean.TypeChecker.RecM.WF.mono, Lean4Lean.TypeChecker.RecM.WF.get, Lean4Lean.TypeChecker.RecM.WF.getEnv, Lean4Lean.TypeChecker.RecM.WF.getLCtx, Lean4Lean.TypeChecker.RecM.WF.readThe, Lean4Lean.TypeChecker.RecM.WF.getNGen, Lean4Lean.TypeChecker.RecM.WF.stateWF, Lean4Lean.TypeChecker.RecM.WF.toLBoolM}

  \lead{In short} Quantifies \texttt{M.WF} over every table satisfying \texttt{Methods.WF}, so a
  \texttt{RecM} proof may \hl{assume its recursive calls sound}, plus a small program logic

\end{lemma}
\begin{proof}
  \leanok
  \lead{Idea} Each combinator unfolds the monad stack on the matching \texttt{M.WF} one.
\end{proof}

\begin{theorem}[Binder introduction preserves the invariant]
  \label{thm:vtc-withlocaldecl}
  \lean{Lean4Lean.TypeChecker.RecM.WF.withLocalDecl, Lean4Lean.TypeChecker.M.WF.withLocalDecl, Lean4Lean.TypeChecker.RecM.WF.withLetDecl}
  \leanok
  \uses{def:vtc-mlctx-wf, def:vtc-vstate-wf, def:vtc-recm-wf}
  \srcloc{Lean4Lean/Verify/TypeChecker/Basic.lean}{439}
  \thesisref{typesys.tex, Weakening (\texttt{thm:weak})}

  \lead{In short} Extending an \texttt{MLCtx} by a fresh binder is \hl{sound}: if \texttt{m} is
  well-formed, \texttt{ty} translates to a type, and \texttt{f (.fvar id)} satisfies \texttt{Q}
  at \texttt{m.vlam id name ty ty' bi} for every unreserved \texttt{id}, then
  \texttt{withLocalDecl name bi ty f} satisfies \texttt{Q} at \texttt{m}. \texttt{withLetDecl} is
  the \texttt{vlet} analogue.
\end{theorem}
\begin{proof}
  \uses{def:vtc-vstate-wf}
  \leanok
  \lead{Idea} The fresh id is reserved by \texttt{s.next} but not \texttt{s}, giving the missing
  \texttt{MLCtx.WF} clause; the invariant is rebuilt at \texttt{s.next}.
\end{proof}

  \label{def:vtc-mkbinders}
  \lean{Lean4Lean.TypeChecker.MLCtx.mkForall, Lean4Lean.TypeChecker.MLCtx.mkForall', Lean4Lean.TypeChecker.MLCtx.mkLambda, Lean4Lean.TypeChecker.MLCtx.mkLambda', Lean4Lean.TypeChecker.MLCtx.mkLetArg, Lean4Lean.TypeChecker.MLCtx.mkLet, Lean4Lean.TypeChecker.AllAbove, Lean4Lean.TypeChecker.AllAbove.wf}

  \lead{In short} Turning the last $n$ free variables back into \hl{binders}: \texttt{mkForall}

\begin{lemma}[Abstraction commutes with translation]
  \label{thm:vtc-mkbinder-tr}
  \lean{Lean4Lean.TypeChecker.MLCtx.WF.mkForall_trS, Lean4Lean.TypeChecker.mkForall_hasType, Lean4Lean.TypeChecker.MLCtx.WF.mkForall'_congr, Lean4Lean.TypeChecker.MLCtx.WF.mkForall_tr, Lean4Lean.TypeChecker.MLCtx.WF.mkLambda_trS, Lean4Lean.TypeChecker.MLCtx.WF.mkLambda'_congr, Lean4Lean.TypeChecker.MLCtx.WF.mkLambda_tr, Lean4Lean.TypeChecker.MLCtx.WF.mkLet_trS}

  \uses{def:vtc-mkbinders, def:vtc-mlctx-wf, ind:trexprs}
  \srcloc{Lean4Lean/Verify/TypeChecker/Basic.lean}{657}
  \thesisref{typesys.tex, Weakening and Substitution}

  \lead{In short} \hl{Abstraction and translation commute}: if a body translates in the full
  context with a model type, its $n$-fold abstraction translates, dropped by $n$, to the
  matching abstraction and is itself a type; primed \texttt{congr} lemmas transport a model

\end{lemma}
\begin{proof}
  \uses{thm:trexprs-abstract}
  \leanok
  \lead{Idea} Induction on $n$: \texttt{vlam} applies \texttt{forallE} to \texttt{abstract};
  \texttt{vlet} builds \texttt{letE}, or rewrites \texttt{lowerLooseBVars} when unused.
\end{proof}

  \lead{In short} Every entry point the declaration checker calls, at \texttt{M} level: \hl{the

\section*{Caveats}
\begin{itemize}
\item Six \texttt{sorry}s: upstream's Lemma~\ref{thm:vtc-reduceprojcore},
  Theorem~\ref{thm:vtc-reducerecursor}, Lemma~\ref{thm:vtc-tryetastruct},
  Lemma~\ref{thm:vtc-isdefequnitlike}; Theorem~\ref{thm:vtc-inferproj-struct}
  (\texttt{trproj}) and Theorem~\ref{thm:vtc-inferproj} (\texttt{mixed}).
\item Theorem~\ref{thm:vtc-iota-reduce-core} (\texttt{iota}) proves the pure $\iota$ step with
  no consumer yet, and a hypothesis stronger than the kernel's call site.
\item Theorem~\ref{thm:vtc-inferproj-struct} (\texttt{trproj}) is unused; its hypotheses are
  kernel metadata, not \texttt{TrProjCtor} itself, the unproved link.
\item Theorem~\ref{thm:vtc-inferproj}, on Theorem~\ref{thm:vtc-infertype-prime}'s import path, is
  documented unprovable for reflexive, indexed, nested single-constructor structures.
\item Every result through \texttt{TrExprS.uniq} carries \texttt{sorryAx} via open
  \texttt{TrProj.uniq} (\srcloc{Lean4Lean/Verify/Typing/Lemmas.lean}{995}).
\item \texttt{checkType.WF} assumes only \texttt{e}'s free variables in scope, never that
  \texttt{e} is translatable; every caller inherits the projection gap.
\end{itemize}
