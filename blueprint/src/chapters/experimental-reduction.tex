\section{The abstract typing interface and proof irrelevance}

The \texttt{VExpr} cluster axiomatizes what Church-Rosser needs as \texttt{Typing}, then builds
proof irrelevance $\equiv_p$ on top.

\begin{definition}[Pointwise definitionally-equal contexts]
\label{inductive:expb-isdefeqctx}
\lean{Lean4Lean.IsDefEqCtx}
\leanok
\srcloc{Lean4Lean/Experimental/NormalEq.lean}{11}

\lead{In short} The \hl{context-conversion relation} used by the \texttt{DFC} lemmas below.
\begin{itemize}
\item \texttt{zero}: relates a prefix $\Gamma_0$ to itself.
\item \texttt{succ}: extends a related pair $\Gamma_1,\Gamma_2$ by hypotheses $A_1,A_2$ with
\texttt{IsDefEqU}~$\Gamma_1\,A_1\,A_2$.
\end{itemize}
\end{definition}

\label{thm:expb-normaleq-trans}
\lean{Lean4Lean.NormalEq.trans, Lean4Lean.NormalEq.symm}

\srcloc{Lean4Lean/Experimental/NormalEq.lean}{451}

\begin{proof}
\uses{family:expb-normaleq-struct, thm:expb-normaleq-defeq, structure:expb-typing}

retype via \texttt{TY.forallEDF}.
\end{proof}

\section{Parallel reduction and Church-Rosser}

Parallel reduction $\gg_\kappa$ folds $\beta$ and the pattern rules into one relation, proved
Church-Rosser up to $\equiv_p$ for the interface, then the concrete judgment.

\begin{definition}[Parallel reduction \texttt{ParRed}]
\label{inductive:expb-parred}
\lean{Lean4Lean.ParRed}
\leanok
\uses{structure:expb-typing, inductive:pattern-matches, inductive:pattern-rhs}
\srcloc{Lean4Lean/Experimental/ParallelReduction.lean}{18}
\thesisref{unique.tex \S\texttt{sec:church\_rosser}: rules for $\gg_\kappa$; $\iota$/$K^+$ fold into
\texttt{extra}.}

\lead{In short} $\Gamma\vdash e\gg_\kappa e'$ is \hl{reflexive and congruent everywhere}, plus
$\beta$ and one abstract pattern rule.
\begin{itemize}
\item \texttt{beta}: $(\lambda x{:}A.\,e_1)\,e_2 \to e_1'[e_2'/x]$ after reducing both parts.
\item \texttt{extra}: given \texttt{TY.Pat p r}, a match \texttt{p.Matches}~$e\,m_1\,m_2$ whose side
conditions \texttt{r.2.OK} hold, and $m_2\,a\gg_\kappa m_2'\,a$ for every captured metavariable, $e$
reduces to \texttt{r.1.apply}~$m_1\,m_2'$.
\end{itemize}
\end{definition}

\begin{definition}[Reducible terms \texttt{NonNeutral}]
\label{def:expb-nonneutral}
\lean{Lean4Lean.NonNeutral}
\leanok
\uses{structure:expb-typing}
\srcloc{Lean4Lean/Experimental/ParallelReduction.lean}{29}
\thesisref{unique.tex \S\texttt{sec:church\_rosser}: compatibility rules apply only where no
substantive rule does.}

\lead{In short} \texttt{NonNeutral TY}~$\Gamma\,e$ holds when $e$ is \hl{a $\beta$ redex or matches
an abstract rule} with its side conditions satisfied.
\end{definition}

\begin{definition}[Complete parallel reduction \texttt{CParRed}]
\label{inductive:expb-cparred}
\lean{Lean4Lean.CParRed}
\leanok
\uses{inductive:expb-parred, def:expb-nonneutral}
\srcloc{Lean4Lean/Experimental/ParallelReduction.lean}{33}
\thesisref{unique.tex \S\texttt{sec:church\_rosser}: the definition of $\ggg_\kappa$.}

\lead{In short} $\ggg_\kappa$ is $\gg_\kappa$ with its \hl{constant and application congruences
guarded} by $\lnot$\texttt{NonNeutral}, so a substantive rule always fires first.
\begin{itemize}
\item \lead{Caveat} The $\lambda$ and $\forall$ congruences stay unguarded, and two overlapping
pattern rules can both fire: the relation is deterministic only up to \texttt{pat\_uniq}.
\end{itemize}
\end{definition}

\begin{lemma}[Basic properties of the two reductions]
\label{family:expb-parred-struct}
\lean{Lean4Lean.ParRed.rfl, Lean4Lean.ParRed.weakN, Lean4Lean.ParRed.instN, Lean4Lean.ParRed.defeq, Lean4Lean.ParRed.hasType, Lean4Lean.ParRed.defeqDFC, Lean4Lean.ParRed.apply_pat, Lean4Lean.ParRed.weakN_inv, Lean4Lean.CParRed.toParRed, Lean4Lean.CParRed.exists, Lean4Lean.Pattern.RHS.apply_liftN}
\leanok
\uses{inductive:expb-parred, inductive:expb-cparred, structure:expb-typing}
\srcloc{Lean4Lean/Experimental/ParallelReduction.lean}{48}
\thesisref{unique.tex, ``properties of $\gg$'': reflexivity, defeq, substitution,
$\ggg\!\Rightarrow\!\gg$, existence.}

\lead{In short} $\gg_\kappa$ is \hl{well behaved}: reflexive, stable under weakening/substitution/%
context conversion (\texttt{rfl}/\texttt{weakN}/\texttt{instN}/\texttt{defeqDFC}, reflecting
weakening via \texttt{weakN\_inv}), sound (\texttt{defeq}/\texttt{hasType}), and total
(\texttt{CParRed.toParRed} refines it, \texttt{CParRed.exists} gives every term a complete reduct).
\end{lemma}
\begin{proof}
\uses{structure:expb-typing, def:expb-nonneutral}
\leanok
\lead{Idea} Each is induction on the reduction, \texttt{extra} via \texttt{Pattern.RHS.apply\_liftN}

\end{proof}

\begin{theorem}[Triangle lemma]
\label{thm:expb-parred-triangle}
\lean{Lean4Lean.ParRed.triangle}
\leanok
\uses{inductive:expb-parred, inductive:expb-cparred, inductive:expb-normaleq}
\srcloc{Lean4Lean/Experimental/ParallelReduction.lean}{282}
\thesisref{unique.tex, the triangle lemma.}

\lead{In short} A complete reduct \hl{absorbs} any single parallel step, up to $\equiv_p$: given
$\Gamma\vdash e:A$, $\Gamma\vdash e\gg_\kappa e'$ and $\Gamma\vdash e\ggg_\kappa e^\bullet$, some
$e^\circ$ has $\Gamma\vdash e'\gg_\kappa e^\circ$ and $\Gamma\vdash e^\circ\equiv_p e^\bullet$.
\end{theorem}
\begin{proof}
\uses{family:expb-parred-struct, family:expb-normaleq-struct, structure:expb-typing}
\leanok
\lead{Idea} Well-founded recursion on the subject, then induction on the complete reduction with
inversion on the parallel step: $\beta$ cases use \texttt{ParRed.instN}; cases where both sides fire
a pattern rule align by \texttt{pat\_uniq}; cases where only one fires are excluded by the
\texttt{NonNeutral} guards.
\end{proof}

\section{Stratified judgments}

Stratification breaks a judgment into layers alternating an abstract typing parameter with a
concrete rule, avoiding the circularity between unique typing and Church-Rosser; typed and
type-erased versions get the same decomposition.

\begin{definition}[Definitional inversion predicate \texttt{DefInv}]
\label{def:expb-definv}
\lean{Lean4Lean.VEnv.DefInv, Lean4Lean.VEnv.DefInv.symm}
\leanok
\uses{def:isdefeq}
\srcloc{Lean4Lean/Experimental/Stratified.lean}{9}
\thesisref{unique.tex: definitional inversion, preceding the unique-typing theorem.}

\lead{In short} \texttt{DefInv env U}~$\Gamma\,e_1\,e_2$ is a \hl{head-shape compatibility}
condition on two types.
\begin{itemize}
\item two $\forall$'s: definitionally equal domains, then codomains in the extended context;
\item two sorts: equivalent levels;
\item sort against $\forall$: \texttt{False}; every other pair: unconstrained.
\end{itemize}
\lead{Caveat} \texttt{DefInv.symm} is proved (symmetric over an ordered environment); nothing in the
file consumes \texttt{DefInv} itself.
\end{definition}

\begin{definition}[One typing layer \texttt{HasType1}]
\label{inductive:expb-hastype1}
\lean{Lean4Lean.VEnv.HasType1}
\leanok
\uses{struct:venv}
\srcloc{Lean4Lean/Experimental/Stratified.lean}{30}
\thesisref{unique.tex \S\texttt{sec:unique}: the definition of $\Gamma\vdash_n e:\alpha$.}

\lead{In short} \texttt{HasType1 env uvars IsDefEq1} is typing whose conversion rule appeals to an
\hl{abstract parameter} \texttt{IsDefEq1} instead of real definitional equality.
\begin{itemize}
\item variable, constant, sort, application, abstraction, product: as usual;
\item \texttt{defeq}: \texttt{IsDefEq1}~$\Gamma\,A\,B\,(\mathtt{.sort}\ u)$. Each such parameter
marks one alternation between typing and conversion.
\end{itemize}
\end{definition}

\begin{definition}[One definitional-equality layer \texttt{IsDefEq1}, with the $\iota$ rule]
\label{inductive:expb-isdefeq1}
\lean{Lean4Lean.VEnv.IsDefEq1, Lean4Lean.VEnv.IsDefEq1.pat}
\leanok
\uses{inductive:expb-hastype1, struct:venv, inductive:pattern-matches, def:pattern-check-realizes}
\contrib{mixed}\srcloc{Lean4Lean/Experimental/Stratified.lean}{46}
\thesisref{unique.tex \S\texttt{sec:unique} (the stratification) with the $\iota$ rule of
\S\texttt{sec:kappa}.}

\lead{In short} \texttt{IsDefEq1} is definitional equality where \hl{typing appeals to
\texttt{HasType1}}; congruence, symmetry, transitivity recurse; \texttt{defeqDF} appeals to the
abstract \texttt{defEq}.
\begin{itemize}
\item \texttt{pat} (added by \texttt{iota}): from \texttt{env.pats p r}, a match
\texttt{p.Matches}~$e\,m_1\,m_2$, a typing $\Gamma\vdash e:A$, a realizer
\texttt{r.2.Realizes}~$m_1\,m_2\,\mathit{chk}$, and $\Gamma\vdash t.1\equiv t.2.1:t.2.2$ for every
$t$ in $\mathit{chk}$, concludes $\Gamma\vdash e\equiv \mathtt{r.1.apply}\,m_1\,m_2:A$. The typing

\begin{lemma}[Concrete domain from shapes, and monotonicity of the interpretation]

\lead{In short} \texttt{D} is a downward-closed, join-complete set of shapes with function space
\texttt{DF} ($\lambda$/\texttt{unlam}/$\Pi$ via \texttt{D.lam}/\texttt{D.pi}); \texttt{interp}
evaluates syntax into it; \texttt{DefEqF} interprets equality and is \hl{monotone} in both shapes.
\end{lemma}

\uses{family:expb-thierry2-shape}
\stSorry{} \texttt{D.pi} and \texttt{DF.mk'} are literal \texttt{sorry}; the interpreted judgment
\texttt{DefEq} is itself an \texttt{axiom}.
\lead{Idea} The \texttt{mono}/\texttt{mono\_r} theorems induct mutually on shape levels via the
(partly \texttt{sorry}) shape lattice; the family is tainted regardless of the textual split.

\section*{Caveats}

\begin{itemize}
\item Every module here is outside \texttt{lake}'s default target; only CI keeps them building,
which is why several accumulate near-duplicate code.
\item \ref{thm:expb-isdefequ-church-rosser} and \ref{thm:expb-induction1}, the two theorems
\texttt{iota} extended, are conditional on pre-existing \texttt{sorry}s
(\texttt{ParallelReduction.lean:699,718}; \texttt{Stratified.lean:96}).
\item The \texttt{SExpr} declarative equality (\ref{inductive:expb-sexpr-isdefeq}) has no $\iota$
rule: its unique-typing results do not model the reduction rules the rest of the chapter has.
\item \texttt{Thierry.lean} and \texttt{Thierry2.lean} declare \texttt{axiom mySorry : $\alpha$}, an
inhabitant of every type: inconsistent by construction.
\item \texttt{WHRed.determ} (\ref{family:expb-sexpr-operational}) is proved only for the
$\beta$/application fragment; every case where a pattern step meets anything is \texttt{sorry}.
\item Some \texttt{sorry}-free-looking files hide gaps: \texttt{Dom.out}'s \texttt{stop} tactic
(\ref{family:expb-domaintheory}) and \texttt{MoreStepIndexed.lean}'s \texttt{\#exit}
(\ref{family:expb-morestepindexed-logrel}) leave declarations unproved or uncompiled.
\end{itemize}
