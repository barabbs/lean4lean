\section{Syntactic shapes of an inductive block}

Predicates here live in namespace \texttt{VExpr} and read an elaborated type.
\texttt{bvarsDesc lo n} is the descending list $[\mathsf{bvar}(lo+n-1),\ldots,\mathsf{bvar}\,lo]$;
\texttt{piArity}/\texttt{piBinders}/\texttt{piBody} read a $\Pi$-telescope and
\texttt{lamArity}/\texttt{lamBody} a $\lambda$-telescope.

\begin{definition}[Occurrence of a block's type formers, \texttt{MentionsConst}]
\label{def:ind-mentions-const}
\lean{Lean4Lean.VExpr.MentionsConst, Lean4Lean.VExpr.mentionsConst, Lean4Lean.VExpr.mentionsConst_iff}
\leanok
\contrib{iota} \srcloc{Lean4Lean/Theory/Inductive.lean}{24}
\uses{ind:vexpr}
\thesisref{axioms.tex \S2.6.1.}

\lead{In short} \hl{Whether $e$ mentions any former in $cs$}: false at $\mathsf{bvar}$/$\mathsf{sort}$,
list membership at $\mathsf{const}$, disjunction at the three binary nodes.
\begin{itemize}
\item \lead{Lean} \texttt{mentionsConst} is the Boolean twin; \texttt{mentionsConst\_iff}
relates them.
\item Mirrors the kernel's \texttt{hasIndOcc} \srcloc{Lean4Lean/Inductive/Add.lean}{118}.
\end{itemize}
\end{definition}

\begin{definition}[Result type of a constructor, \texttt{CtorResult}]
\label{def:ind-ctor-result}
\lean{Lean4Lean.VExpr.CtorResult}
\leanok
\contrib{iota} \srcloc{Lean4Lean/Theory/Inductive.lean}{44}
\uses{def:pi-telescope, def:bvarsdesc, def:mkapps}
\thesisref{axioms.tex \S2.6.1.}

\lead{In short} $ty.\mathsf{CtorResult}\;T\,np\,nf\,nind$ holds when \hl{$ty.\mathsf{piArity}
= np+nf$ and $ty.\mathsf{piBody} = (\mathsf{const}\,T\,us).\mathsf{mkApps}(\texttt{bvarsDesc}\,nf\,np
\mathbin{+\!\!+} idx)$} for some levels $us$ and index list $idx$ of length $nind$.
\end{definition}

\begin{definition}[A valid recursive occurrence of the block, \texttt{ValidIndApp}]
\label{def:ind-valid-ind-app}
\lean{Lean4Lean.VExpr.ValidIndApp}
\leanok
\contrib{iota} \srcloc{Lean4Lean/Theory/Inductive.lean}{75}
\uses{def:ind-mentions-const, def:bvarsdesc, def:mkapps}
\thesisref{axioms.tex \S2.6.1.}

\lead{In short} $e.\mathsf{ValidIndApp}\;fs\;np\;d$ holds when
\hl{$e = (\mathsf{const}\;T\;us).\mathsf{mkApps}(\texttt{bvarsDesc}\;d\;np \mathbin{+\!\!+} idx)$}
for some $T \in fs$, $us$, and $idx$ with no member mentioning $fs$; $d$ is the offset from
the field level.
\begin{itemize}
\item \lead{Caveat} The kernel's \texttt{isValidIndAppIdx}
\srcloc{Lean4Lean/Inductive/Add.lean}{157} also checks exact arity; this predicate drops
it, accepting an over-applied former (later rejected by \texttt{WF.universes}).
\end{itemize}
\end{definition}

\begin{definition}[Strict positivity of one constructor field, \texttt{FieldPositive}]
\label{def:ind-field-positive}
\lean{Lean4Lean.VExpr.FieldPositive}
\leanok
\contrib{iota} \srcloc{Lean4Lean/Theory/Inductive.lean}{93}
\uses{def:ind-valid-ind-app, def:ind-mentions-const, def:pi-telescope}
\thesisref{axioms.tex \S2.6.1.}

\lead{In short} $ty$ is $\mathsf{FieldPositive}$ at depth $d$ iff
\hl{$\neg\mathsf{MentionsConst}\;fs\;ty$, or every $\Pi$-binder of $ty$ avoids $fs$ and
$ty.\mathsf{piBody}.\mathsf{ValidIndApp}\;fs\;np\;(d+ty.\mathsf{piArity})$}: the block is
absent, or reached only through a valid recursive occurrence.
\begin{itemize}
\item \lead{Caveat} Reads manifest $\Pi$-binders; the kernel's \texttt{checkPositivity}
\srcloc{Lean4Lean/Inductive/Add.lean}{184} reduces to whnf first, so this is stricter.
\item The thesis also forbids a recursive argument from being referred to by later
arguments; this predicate, like the kernel, does not enforce that.
\end{itemize}
\end{definition}

\begin{definition}[Strict positivity of a constructor, \texttt{CtorPositive}]
\label{def:ind-ctor-positive}
\lean{Lean4Lean.VExpr.CtorPositive}
\leanok
\contrib{iota} \srcloc{Lean4Lean/Theory/Inductive.lean}{101}
\uses{def:ind-field-positive, def:ind-mentions-const, def:pi-telescope}

\lead{In short} $ty.\mathsf{CtorPositive}\;fs\;np$: \hl{no former occurs in the $np$
parameter binders, and for $i < ty.\mathsf{piArity}-np$ the binder at $np+i$ is
\texttt{FieldPositive} at depth $i$}.
\end{definition}

\begin{definition}[A field appears among the result indices, \texttt{FieldInIndices}]
\label{def:ind-field-in-indices}
\lean{Lean4Lean.VExpr.FieldInIndices}

\lead{In short} $ty.\mathsf{FieldInIndices}\;np\;i$ holds when
\hl{$\mathsf{bvar}\,(ty.\mathsf{piArity}-np-1-i)$ occurs in
$ty.\mathsf{piBody}.\mathsf{getAppArgs}.\mathsf{drop}\;np$}.
\begin{itemize}
\item The kernel's \texttt{isLargeEliminator} \srcloc{Lean4Lean/Inductive/Add.lean}{275}
searches all result arguments; agreement relies on \texttt{CtorResult} pinning the first
$np$ to values $\ge nf$.
\end{itemize}
\end{definition}

\begin{definition}[The de Bruijn context of a constructor field, \texttt{fieldCtx}]

\end{definition}

\begin{definition}[Type of a recursor's major premise, \texttt{MajorApp}]
\label{def:ind-major-app}
\lean{Lean4Lean.VExpr.MajorApp}
\leanok

\lead{In short} $A.\mathsf{MajorApp}\;T\;np\;nm\;nmin\;nind$ holds when
\hl{$A = (\mathsf{const}\;T\;us).\mathsf{mkApps}(\texttt{bvarsDesc}\,(nm{+}nmin{+}nind)\;np
\mathbin{+\!\!+} \texttt{bvarsDesc}\;0\;nind)$} for some $us$: $T$ applied to the parameters and

\end{definition}

\begin{definition}[Shape of a motive binder, \texttt{MotiveShape}]
\label{def:ind-motive-shape}
\lean{Lean4Lean.VExpr.MotiveShape}
\leanok
\contrib{iota} \srcloc{Lean4Lean/Theory/Inductive.lean}{121}
\uses{def:headconst, def:pi-telescope}
\thesisref{axioms.tex \S2.6.3.}

\lead{In short} A motive type is a \hl{$\Pi$-telescope ending in a sort, whose last binder
is headed by a constant} (\texttt{motiveFormer?}); a bare \texttt{VEnv} has no notion of
type former, so \texttt{RecShape} ties the eliminated motive's head to the major premise.
\end{definition}

\begin{definition}[A minor premise targets one of the motives, \texttt{MinorHeaded}]
\label{def:ind-minor-headed}
\lean{Lean4Lean.VExpr.MinorHeaded}
\leanok
\contrib{iota} \srcloc{Lean4Lean/Theory/Inductive.lean}{128}
\uses{def:app-spine, def:pi-telescope}
\thesisref{axioms.tex \S2.6.3.}

\lead{In short} $A.\mathsf{MinorHeaded}\;i\;nm$ holds when
\hl{$A.\mathsf{piBody}.\mathsf{getAppFn} = \mathsf{bvar}\,(A.\mathsf{piArity}+i+k)$} for some
$k<nm$, i.e.\ minor $i$'s body is headed by motive $nm-1-k$, not tied to the constructor's
own former.
\end{definition}

\begin{definition}[The minor premise belonging to a constructor, \texttt{MinorFor}]
\label{def:ind-minor-for}
\lean{Lean4Lean.VExpr.MinorFor}
\leanok
\contrib{iota} \srcloc{Lean4Lean/Theory/Inductive.lean}{134}
\uses{def:recheaded, def:headconst, def:app-spine}
\thesisref{axioms.tex \S2.6.3.}

\lead{In short} $A.\mathsf{MinorFor}\;c$: \hl{$A$ is \texttt{RecHeaded} and its last argument
is headed by constructor $c$}.
\end{definition}

\begin{definition}[Shape of a recursor type, \texttt{RecShape}]
\label{def:ind-rec-shape}
\lean{Lean4Lean.VExpr.RecShape}
\leanok
\contrib{iota} \srcloc{Lean4Lean/Theory/Inductive.lean}{143}
\uses{def:ind-motive-shape, def:ind-minor-headed, def:ind-major-app, def:pi-telescope, def:mkapps, def:bvarsdesc}
\thesisref{axioms.tex \S2.6.3.}

\lead{In short} $ty.\mathsf{RecShape}\;np\;nm\;nmin\;nind$: \hl{parameters, then $nm$
motives, then $nmin$ minors, then the major premise, then the eliminated motive applied to
the indices and the major}.
\begin{itemize}
\item $ty.\mathsf{piArity} = np+nm+nmin+nind+1$; binders $np..np{+}nm{-}1$ are
\texttt{MotiveShape}; the next $nmin$ are \texttt{MinorHeaded}; the last binder is
\texttt{MajorApp} for the type former $T$ heading motive $j$'s last binder ($j<nm$); and
$ty.\mathsf{piBody} = \mathsf{bvar}\,(nind+nmin+1+(nm-1-j))$ applied to
\texttt{bvarsDesc 0 (nind+1)}.
\item Parameter and index binder types, and the heads of non-eliminated motives, are
unconstrained: a weakening of $\kappa$ and $\varepsilon$.
\end{itemize}
\end{definition}

\begin{definition}[Shape of a constructor type, \texttt{CtorShape}]
\label{def:ind-ctor-shape}
\lean{Lean4Lean.VExpr.CtorShape}
\leanok

\thesisref{no direct counterpart.}

\lead{In short} $ty.\mathsf{CtorShape}\;arity$: \hl{$ty$ has $arity$ binders and is
\texttt{CtorHeaded}}, naming neither the type former nor the index count.
\begin{itemize}
\item Weaker than \texttt{CtorResult}; its two halves are used separately
(\texttt{PatsIota.ctor\_shape} only \texttt{CtorHeaded}, the front end
\srcloc{Lean4Lean/Verify/Environment/Basic.lean}{514} only the arity).
\end{itemize}
\end{definition}

\begin{definition}[The type former a recursor eliminates, \texttt{majorFormer?}]
\label{def:ind-major-former}
\lean{Lean4Lean.VExpr.majorFormer?}
\leanok

\lead{In short} $ty.\mathsf{majorFormer?}\;idx = ty.\mathsf{piBinders}[idx]?.\mathsf{bind}\;\mathsf{headConst?}$:
\hl{the head constant of binder $idx$}, the major premise when $idx=r.\mathsf{getMajorIdx}$.
\end{definition}

\begin{definition}[Shape of an iota-rule reduct, \texttt{RuleShape}]
\label{def:ind-rule-shape}
\lean{Lean4Lean.VExpr.RuleShape}
\leanok
\contrib{iota} \srcloc{Lean4Lean/Theory/Inductive.lean}{174}
\uses{def:lam-telescope, def:bvarsdesc, def:mkapps}
\thesisref{axioms.tex \S2.6.4, $\mathrm{rec}_P\;C\;e\;p[b]\;(c\;b) \equiv e_c\;b\;v$.}

\lead{In short} $rhs.\mathsf{RuleShape}\;np\;nm\;nmin\;nf\;nrec\;j$ holds when
$rhs.\mathsf{lamArity} = np+nm+nmin+nf$ and, for some $recArgs$ of length $nrec$,
\hl{$rhs.\mathsf{lamBody} = \mathsf{bvar}\,(nf+(nmin-1-j))$ applied to
$\texttt{bvarsDesc}\;0\;nf \mathbin{+\!\!+} recArgs$}: minor $j$ applied $\eta$-long to the
fields, then $nrec$ more arguments.
\begin{itemize}
\item \lead{Caveat} Pins the recursive calls $v_i$ only in number, not in the shape
$\lambda x{::}\xi_i.\;\mathrm{rec}_P\;C\;e\;\pi_i[b,x]\;(u_i\;x)$ the thesis requires;
\texttt{WF.rules\_wf} only forces them well-typed.
\end{itemize}
\end{definition}

neither proved here.

syntactic test.

\section{Patterns, matching and schematic rule reducts}

This section is \srcloc{Lean4Lean/Theory/Typing/Pattern.lean}{1}. The first five nodes are
upstream's abstract schematic-rewrite machinery, extended additively for the $\iota$ rule.

\begin{definition}[Patterns, subpatterns and pattern intersection]
\label{inductive:pattern-core}
\lean{Lean4Lean.Pattern, Lean4Lean.Pattern.varN, Lean4Lean.Subpattern, Lean4Lean.Subpattern.trans, Lean4Lean.Subpattern.sizeOf_le, Lean4Lean.Subpattern.antisymm, Lean4Lean.Arity, Lean4Lean.Arity.subpattern, Lean4Lean.Pattern.inter, Lean4Lean.Pattern.inter_self, Lean4Lean.Pattern.inter_comm, Lean4Lean.Pattern.LE}
\leanok
\srcloc{Lean4Lean/Theory/Typing/Pattern.lean}{8}
\uses{ind:vexpr}
\lead{In short} A \texttt{Pattern} is a constant, an application of two patterns, or a
\hl{\texttt{var} hole} over a pattern (\texttt{varN} iterates \texttt{var}).
\begin{itemize}
\item \texttt{Subpattern}: reflexive structural containment (antisymmetric via
\texttt{sizeOf}); \texttt{Arity}: argument positions above a subpattern, used only in the
experimental development \srcloc{Lean4Lean/Experimental/ShapeLogRel.lean}{3436}
\srcloc{Lean4Lean/Experimental/ShapeLogRel.lean}{3620}.
\item \texttt{Pattern.inter}: common refinement of two patterns; a \texttt{var} hole matches
any argument, so at an \texttt{app}/\texttt{var} pair the \texttt{app}'s argument survives.
\end{itemize}
\end{definition}

\begin{definition}[Matching a pattern against an expression]
\label{inductive:pattern-matches}
\lean{Lean4Lean.Pattern.Path, Lean4Lean.Pattern.Matches, Lean4Lean.Pattern.Matches.uniq, Lean4Lean.Pattern.matches_determ, Lean4Lean.Pattern.matches_inter, Lean4Lean.Pattern.OnArgs}
\leanok
\srcloc{Lean4Lean/Theory/Typing/Pattern.lean}{132}
\uses{inductive:pattern-core}
\lead{In short} $p.\mathsf{Path}$ enumerates $p$'s holes; \hl{$p.\mathsf{Matches}\;e\;ls\;g$}
says $e$ matches $p$ at levels $ls$ with hole assignment $g$.
\begin{itemize}
\item Matching is deterministic; $e$ matches $p$ and $q$ iff it matches their intersection.
\item \lead{Caveat} In the \texttt{app} case the argument side's level list is discarded, so
an $\iota$ redex constrains the recursor's universes but not the constructor's.
\end{itemize}
\end{definition}

\begin{definition}[Rule reducts and side conditions]
\label{inductive:pattern-rhs}
\lean{Lean4Lean.Pattern.RHS, Lean4Lean.Pattern.Check, Lean4Lean.Pattern.RHS.apply, Lean4Lean.Pattern.Check.OK, Lean4Lean.Pattern.Check.OK.map}
\leanok
\srcloc{Lean4Lean/Theory/Typing/Pattern.lean}{155}
\uses{inductive:pattern-core, inductive:pattern-matches}
\lead{In short} A $p.\mathsf{RHS}$ is a tree of \hl{closed fixed terms, applications, and
references to $p$'s holes}; \texttt{RHS.apply} instantiates it at a match.
\begin{itemize}
\item A $p.\mathsf{Check}$ is a list of reduct pairs to be related; \texttt{Check.OK}
asserts them against a relation; \texttt{OK.map} transports it along a relation morphism.
\end{itemize}
\end{definition}

\begin{lemma}[Lifting and instantiation of matches and reducts]
\label{family:pattern-transport-master}
\lean{Lean4Lean.Pattern.RHS.lift'_apply, Lean4Lean.Pattern.RHS.liftN_apply, Lean4Lean.Pattern.matches_lift', Lean4Lean.Pattern.matches_liftN, Lean4Lean.Pattern.RHS.instN_apply, Lean4Lean.Pattern.matches_instN}
\leanok
\srcloc{Lean4Lean/Theory/Typing/Pattern.lean}{186}
\uses{inductive:pattern-matches, inductive:pattern-rhs}
\lead{In short} \texttt{RHS.apply} commutes with \hl{lifting and instantiation} (fixed parts
are closed); matching commutes with lifting both ways, with instantiation only forward.
\end{lemma}
\begin{proof}
\leanok
\lead{Idea} Structural induction on the reduct for the \texttt{apply} lemmas; on the
\texttt{Matches} derivation for \texttt{matches\_lift'}, with \texttt{matches\_liftN} a
corollary via \texttt{lift'\_consN\_skipN}.
\end{proof}

\begin{definition}[The two schematic rule shapes, \texttt{SimplePattern}]
\label{inductive:simple-pattern}
\lean{Lean4Lean.SimplePattern, Lean4Lean.SimplePattern.toPattern}
\leanok
\srcloc{Lean4Lean/Theory/Typing/Pattern.lean}{484}
\uses{inductive:pattern-core}
\lead{In short} A \texttt{SimplePattern} is \hl{either an $\iota$ redex or a bare constant}.
\begin{itemize}
\item $\mathsf{iota}\;rec\;M\;ctor\;N$ unfolds to $(\mathsf{const}\;rec).\mathsf{varN}\;M$
applied to $(\mathsf{const}\;ctor).\mathsf{varN}\;N$.
\item $\mathsf{defn}\;head$: the bare constant; never registered by anything, since
$\delta$ rules stay in \texttt{defeqs} (\ref{def:indparams-defeqs-as-pats}).
\end{itemize}
\end{definition}

\begin{lemma}[Combinatorics of constructor spines]
\label{family:pattern-spine-combinatorics}
\lean{Lean4Lean.Pattern.subpattern_varN_const, Lean4Lean.Pattern.not_app_subpattern_varN_const, Lean4Lean.Pattern.varN_const_inter, Lean4Lean.Pattern.varN_const_inj}
\leanok
\contrib{iota} \srcloc{Lean4Lean/Theory/Typing/Pattern.lean}{76}
\uses{inductive:pattern-core}
\thesisref{no direct counterpart.}

\lead{In short} \hl{Spines $(\mathsf{const}\;c).\mathsf{varN}\;k$ behave like arities}: every
subpattern is a shorter spine on $c$, no spine has an application subpattern, and two spines
intersect (or agree) only when head and length match.
\end{lemma}
\begin{proof}
\leanok
\lead{Idea} Induction on $k$ for the subpattern claim, inversion for the no-application
case; \texttt{varN\_const\_inter} by a four-way induction on the two lengths, and
\texttt{varN\_const\_inj} from \texttt{inter\_self}.
\end{proof}

\begin{definition}[The holes a reduct mentions, \texttt{RHS.Uses}]
\label{def:pattern-rhs-uses}

\section*{Caveats}
\label{sec:ind-review}

\begin{itemize}
\item Nothing derives \ref{structure:ind-wf} from the kernel's own inductive checker
(\texttt{Lean4Lean/Inductive/Add.lean}); it remains a hypothesis, checked only by
\texttt{decide} on concrete kernel data (\texttt{Tests/IotaShape.lean}).
\item K-like reduction is not modelled (\ref{def:ind-add-rec-rule} registers only the
constructor rule), so for subsingleton eliminators the model is weaker than the kernel.
\item \ref{structure:ind-wf} excludes nested inductives and reads manifest binders for
positivity and universes, stricter than the kernel's whnf-based check.
\item \ref{def:ind-rule-shape} pins a rule's recursive calls only in number, not in shape, so
\ref{structure:ind-wf} accepts $\iota$ rules the kernel would never generate.
\item \ref{def:indparams-toparams} applies only to environments with no definitional axiom;
the one instance built, \ref{def:indparams-induct-params}, is a single inductive block.
\item Lemma~\ref{thm:indparams-cr-defeq-of-induct} is conditional: \texttt{IsDefEq.church\_rosser}
depends on two \texttt{sorry}s in \texttt{NormalEq.parRed} (see the \texttt{axfoot} above).
\end{itemize}
