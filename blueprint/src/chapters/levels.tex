for all $\rho$.

\section{The canonical form and its semantics}

\begin{definition}[Semantics of the normal form]
\label{def:lvl-normlevel-eval}
\lean{Lean.Level.Normalize.evalParam, Lean.Level.Normalize.VarNode.eval, Lean.Level.Normalize.Node.eval, Lean.Level.Normalize.allNZ, Lean.Level.Normalize.evalPath, Lean.Level.Normalize.NormLevel.eval}
\leanok
\srcloc{Lean4Lean/Verify/Level.lean}{234}
\uses{def:vlevel-eval, def:kernel-level-normalize}
\thesisref{soundness.tex: valuation semantics; \texttt{evalPath} generalises \texttt{imax} to a guard
set.}

\lead{In short} The denotation of a normal form under $\rho$: \hl{a \texttt{NormLevel} is the max, over
its entries, of each entry's guarded value}. \texttt{evalParam}: $ls$-lookup then $\rho$ (else $0$);
\texttt{VarNode.eval} $x+k = \texttt{evalParam}(x)+k$; \texttt{Node.eval}: max of constant and variable
nodes; \texttt{allNZ path}: every name nonzero; \texttt{evalPath path n}: $n$ if so, else $0$.
\end{definition}

\begin{lemma}[Sorted-list subset test]
\label{fam:lvl-subset}
\lean{Lean.Level.Normalize.subset_length, Lean.Level.Normalize.subset_mem, Lean.Level.Normalize.subset_eq, Lean.Level.Normalize.subset_of_sorted, Lean.Level.Normalize.subset_antisymm}
\leanok
\srcloc{Lean4Lean/Verify/Level.lean}{658}

\lead{In short} The merge-style test \texttt{subset cmp l1 l2} \hl{decides list inclusion on sorted
lists}: \texttt{l1} $\subseteq$ \texttt{l2} and $|l1| \le |l2|$, with the converse, equal-length and
antisymmetry facts holding when both are sorted.
\end{lemma}
\begin{proof}

\leanok
\lead{Idea} Induction along \texttt{subset}'s three-way branch, using
\texttt{LawfulBEqCmp.compare\_eq\_iff\_beq} and that a sorted list's head is below every later

\end{proof}

\begin{lemma}[Correctness of one subsumption step]
\label{fam:lvl-subsume-node}

\lead{In short} \texttt{subsumeVars}, \texttt{Node.subsumeBy} and \texttt{Node.subsume}: \hl{subsumption
never changes the total value}. The result is a sublist of the first argument, a dropped variable had a
same-name dominator at a larger offset, and \texttt{subsumeBy\_eval\_iff} bounds \texttt{n1.subsumeBy

\end{lemma}

\leanok
\lead{Idea} Simultaneous induction on the two sorted variable lists; the \texttt{same} flag

\end{proof}

\section{Reconstruction: trees and admissible \texttt{imax} chains}
\label{sec:lvl-reconstruction}

A normal form must become a \texttt{Lean.Level}, choosing an \texttt{imax} chain per key. A chain is
not free -- each edge contributes a sublevel -- so only admissible orders, chosen from the sublevels
alone, may be used.

\begin{definition}[Semantics of the tree]
\label{def:lvl-tree-eval}
\lean{Lean.Level.Normalize.Tree.eval, Lean.Level.Normalize.Tree.evalChild, Lean.Level.Normalize.Tree.eval_eq}
\leanok
\srcloc{Lean4Lean/Verify/Level.lean}{1306}
\uses{def:lvl-normlevel-eval}
\lead{In short} A \texttt{Tree} node evaluates to the max of its own value and its children's, \hl{each
child guarded by \texttt{Nat.imax}} of the child's value and its edge label's.
\begin{itemize}
\item Mirrors exactly the level \texttt{imax (reify t) (param a)} that \texttt{reify} emits for an
edge.
\end{itemize}
\end{definition}

\begin{theorem}[\texttt{reify} preserves value]
\label{thm:lvl-tree-reify-eval}
\lean{Lean.Level.Normalize.Tree.reify_eval, Lean.Level.Normalize.Tree.reifyChild_eval, Lean.Level.Normalize.Tree.plainOffset?_eval, Lean.Level.Normalize.Tree.reifyChild_ge}
\leanok
\srcloc{Lean4Lean/Verify/Level.lean}{1404}
\uses{def:lvl-tree-eval}
\lead{In short} Evaluating the \texttt{Lean.Level} that \texttt{t.reify} emits, at \texttt{evalParam
ls} $\rho$, \hl{gives \texttt{Tree.eval ls} $\rho$ \texttt{t}}.
\end{theorem}
\begin{proof}
\leanok
\lead{Idea} Mutual induction with \texttt{reifyChild\_eval}. The \texttt{plainOffset?} optimisation
drops an \texttt{imax} guard on a child; the two forms differ only at parameter $0$, covered by the
node's own constant (\srcloc{Lean4Lean/Level.lean}{264}).
\end{proof}

\begin{definition}[Subtrees at a path]
\label{def:lvl-tree-at}
\lean{Lean.Level.Normalize.Tree.At, Lean.Level.Normalize.Tree.At.append, Lean.Level.Normalize.Tree.At.suffix, Lean.Level.Normalize.Tree.At.nil_inv, Lean.Level.Normalize.Tree.At.append_inv, Lean.Level.Normalize.Tree.At_modify_self, Lean.Level.Normalize.Tree.At_modify_of_ne, Lean.Level.Normalize.Tree.At_modify_inv}
\leanok
\srcloc{Lean4Lean/Verify/Level.lean}{1449}

\lead{In short} \texttt{Tree.At t p t'}: \texttt{t'} is the subtree reached by following \texttt{p}'s
labels, innermost first -- \hl{the order \texttt{Tree.modify} takes}. A modification is visible at its
own path and invisible elsewhere; every subtree of a modified tree is a subtree of the original or the
modified one.
\end{definition}

\begin{theorem}[Bounded iff all sublevels are]
\label{thm:lvl-tree-eval-le-iff}
\lean{Lean.Level.Normalize.Tree.eval_le_iff, Lean.Level.Normalize.Tree.eval_le_of, Lean.Level.Normalize.Tree.At.le, Lean.Level.Normalize.Tree.At.edge_le}
\leanok
\srcloc{Lean4Lean/Verify/Level.lean}{1595}
\uses{def:lvl-tree-at, def:lvl-tree-eval}
\lead{In short} \texttt{Tree.eval ls} $\rho$ \texttt{t} $\le m$ iff \hl{every subtree's node value and
every edge's own contribution are $\le m$}.
\end{theorem}
\begin{proof}
\uses{fam:lvl-eval-le}
\leanok
\lead{Idea} Forward: \texttt{At.le} and \texttt{At.edge\_le}. Converse: mutual

\begin{definition}[Sublevels and syntactic domination]
\label{def:lvl-sub}
\lean{Lean.Level.Normalize.Sub, Lean.Level.Normalize.Sub.path, Lean.Level.Normalize.Sub.le, Lean.Level.Normalize.Sub.le.trans, Lean.Level.Normalize.Sub.le.antisymm, Lean.Level.Normalize.NormLevel.HasSub, Lean.Level.Normalize.Sub.eval, Lean.Level.Normalize.NormLevel.HasSub.le_eval, Lean.Level.Normalize.Node.HasSub, Lean.Level.Normalize.NormLevel.hasSub_iff}

decides the semantic one.}

\lead{In short} A \texttt{Sub} is one sublevel ($C(p,k)$ or $V(p,x,k)$); \texttt{Sub.le} is \hl{the
paper's domination order} -- dominator's condition set a subset of the dominated one, a constant
dominated by a variable of offset $K$ up to $K+1$, a variable only by itself at a larger offset.
Transitive, antisymmetric on sorted sets. \texttt{HasSub s t}: \texttt{t} recorded in \texttt{s}, below

\begin{theorem}[Separation: semantic forces syntactic]

\end{theorem}

\uses{fam:lvl-eval-le}
\leanok
\lead{Idea} A separating valuation: the sublevel's names to $1$, its variable to $N$ exceeding every
constant/offset of \texttt{l2}, else $0$; the entry exceeding $N$ is the dominator.
\thesisref{The converse direction of axioms.tex's algorithmic inequality, stated there but not
proved.}
\end{proof}

\begin{definition}[Structural comparison of bases]

\leanok

\texttt{getLevelOffset} never exceeds the level itself (\texttt{size}, \texttt{structCmp} private).

\begin{definition}[Ordering-valued comparison of levels]

\leanok

\texttt{getLevelOffset}), then, on equal bases, by \texttt{succ} offset: \hl{the \texttt{Ordering}-valued

\end{definition}

\begin{lemma}[\texttt{baseCmp} is tag then structure]

\leanok

\midrule

\label{thm:eqvmgr-addequiv-wf}
\lean{Lean4Lean.TypeChecker.Inner.addEquiv.WF}
\leanok
\srcloc{Lean4Lean/Verify/EquivManager.lean}{313}
\uses{def:kernel-equiv-manager, ind:trexprs, def:trexpr}
\lead{In short} Updating the type checker's state by \texttt{addEquiv e1 e2} \hl{preserves the whole
\texttt{VState} invariant}, provided \texttt{e1}, \texttt{e2} translate to the same term (one strictly,
one up to defeq).
\end{theorem}
\begin{proof}
\uses{fam:eqvmgr-primitives-wf, thm:vtc-methods-iface}
\leanok
\lead{Idea} Unfold into two \texttt{toNode} calls and a \texttt{merge}; apply \texttt{toNode.WF}
twice and \texttt{merge.WF} once, with \texttt{IsDefEqE.defeq} from weakening both translations.
\end{proof}

\begin{theorem}[Soundness of the memoising wrapper]
\label{thm:eqvmgr-isdefeq-wf}
\lean{Lean4Lean.TypeChecker.Inner.isDefEq.WF}
\leanok
\srcloc{Lean4Lean/Verify/EquivManager.lean}{325}
\uses{ind:trexprs, def:hastype-istype}
\lead{In short} If \texttt{e1}, \texttt{e2} translate to \texttt{e1'}, \texttt{e2'}: running the type
checker's \texttt{isDefEq e1 e2} is sound, and \hl{a \texttt{true} result yields their definitional
equality}.
\end{theorem}
\begin{proof}
\uses{thm:eqvmgr-addequiv-wf, thm:vtc-methods-iface}
\leanok
\lead{Idea} Chain \texttt{isDefEqCore}'s soundness with \texttt{addEquiv.WF} for the caching
branch. Used throughout \texttt{Verify/TypeChecker/} and at
\srcloc{Lean4Lean/Verify/Environment.lean}{178}.
\end{proof}

\section*{Caveats}

\begin{itemize}
\item Two \texttt{Std.TreeMap} axioms (\ref{family:treemap-axioms}) underlie every main normal-form
theorem; results reaching core's \texttt{normalize} add the \texttt{LevelStd} axioms
(\ref{family:level-axioms}).
\item \texttt{RelevantEq.uniq}, \texttt{IsDefEqE.trExpr} and \texttt{IsDefEqE.uniq} carry
\texttt{sorryAx} through the open \texttt{TrProj.uniq} (\ref{thm:trproj-uniq}); \texttt{isDefEq.WF} is
the entry point used throughout the type checker, so this hole sits on the main path, not a corner.
\item Only Theorem~\ref{thm:lvl-isequivlist-wf} reaches the verified type checker; the reconstruction
layer (\texttt{normalize'} and its theorems) is unused by any verified path, and \texttt{geq'} is
called only from unverified code.
\item \texttt{QSort.lean} is vendored upstream, included only because \texttt{Array.qsort} lacks a
standard-library spec.
\item The bit-packing lemmas, part of \texttt{NormLt.lean}, and all of \texttt{QSort.lean} close with
bare \texttt{grind} or \texttt{bv\_decide}, not auditable by hand.
\item \texttt{mkData\_depth}/\texttt{hasParam}/\texttt{hasMVar} are unused and are the only consumers of
the \texttt{mkData\_eq} axiom; \texttt{RelevantEq.symm}/\texttt{trans} are unused too, with no axiom
involved.
\end{itemize}
