\section{Conditions and their model-side reading}

Several primitives (\texttt{Nat.div}, \texttt{Nat.mod}, \texttt{Nat.gcd}, \texttt{Nat.bitwise})
branch on a decidable predicate. A \texttt{Condition} packages the predicate, its decision
procedure and the evidence relating them; the definitions below are what a successful
\texttt{Condition.check} hands back.

\begin{definition}[Model-side shapes of a conditional]
  \label{prim-cond:def:ite-dite-app}
  \lean{Lean4Lean.Primitive.iteApp, Lean4Lean.Primitive.diteApp}
  \leanok
  \uses{ind:vexpr}
  \srcloc{Lean4Lean/Verify/Environment/Primitive/Condition.lean}{40}

  \lead{In short} The \hl{\texttt{VExpr} shape} of a checked \texttt{ite}/\texttt{dite}.
  \begin{itemize}
  \item $\mathtt{iteApp}\,\alpha\,p\,\mathit{inst}\,t\,e$ reads $@\mathtt{ite}.\{1\}\,\alpha\,p\,\mathit{inst}\,t\,e$.
  \item $\mathtt{diteApp}\,p\,\mathit{inst}\,t\,e$ reads $@\mathtt{dite}.\{1\}\,\mathtt{Nat}\,p\,\mathit{inst}\,t\,e$, whose branches take the decision's proof.
  \end{itemize}
\end{definition}

\begin{definition}[A reflection together with its translation, \texttt{Reflection.WF}]
  \label{prim-cond:struct:reflection-wf}
  \lean{Lean4Lean.Primitive.Reflection.WF}
  \leanok
  \uses{ind:trexprs, family:kernel-primitive-dsl}
  \srcloc{Lean4Lean/Verify/Environment/Primitive/Condition.lean}{51}

  \lead{In short} What it means for a source \hl{\texttt{Reflection}} to be checked and translated.
  \begin{itemize}
  \item Carries a \texttt{VExpr} $\mathtt{type}'$ and a translation $c \vdash r.\mathtt{type}
    \leadsto \mathtt{type}'$.
  \item Carries the typing $\mathtt{type}' : \mathtt{Prop} \to \mathtt{Bool} \to \mathtt{Prop}$,
    established by \texttt{Reflection.check}.
  \item The translation is a field, not existential, because \texttt{WF\_ite}/\texttt{WF\_dite}
    are equations about it.
  \end{itemize}
\end{definition}

\begin{definition}[What is known about \texttt{toDec}'s type]
  \label{prim-cond:def:reflection-todect}
  \lean{Lean4Lean.Primitive.Reflection.WF.ToDecT, Lean4Lean.Primitive.Reflection.WF.ToDecT.spine}
  \leanok
  \uses{prim-cond:struct:reflection-wf, fam:strong-inversions}
  \srcloc{Lean4Lean/Verify/Environment/Primitive/Condition.lean}{68}

  \lead{In short} Well-typedness of \hl{\texttt{toDec}} applied to a proof of its own type.
  \begin{itemize}
  \item \texttt{ToDecT toDec'}: $\mathtt{toDec}'\,p\,bb\,H$ is well formed for every $p$, $bb$, $H :
    \mathtt{type}'\,p\,bb$, in every extension and well-formed context.
  \item \texttt{ToDecT.spine} peels that application through its three pi types via
    \texttt{VExpr.WF.app\_inv}, without pinning the domains to $\mathtt{Prop}$, $\mathtt{Bool}$,
    $\mathtt{type}\,p\,bb$.
  \item \lead{Caveat} Not a field of \ref{prim-cond:struct:reflection-wf}: only the two optional
    checks establish it.
  \end{itemize}
\end{definition}

\begin{definition}[The two optional contents of a reflection check]
  \label{prim-cond:def:reflection-wf-ite-dite}
  \lean{Lean4Lean.Primitive.Reflection.WF.WF_ite, Lean4Lean.Primitive.Reflection.WF.WF_dite}
  \leanok
  \uses{prim-cond:struct:reflection-wf, prim-cond:def:reflection-todect}
  \srcloc{Lean4Lean/Verify/Environment/Primitive/Condition.lean}{106}

  \lead{In short} What the two \hl{optional} reflection checks establish.
  \begin{itemize}
  \item \texttt{WF\_ite}: at $H : \mathtt{type}'\,p\,(\mathtt{boolLit}\,b)$, the \texttt{ite} the
    reflection builds on $\mathtt{toDec}'\,p\,b\,H$ is definitionally the branch $b$ selects, for
    every $\alpha$, both branches, any extension and context.
  \item \texttt{WF\_dite}: the same at \texttt{Nat}, branches applied to the proof
    \texttt{ofTrue}/\texttt{ofFalse} extract; those two are existential, since only this check
    types them.
  \item Both quantify over the context, unlike \ref{prim-cond:struct:reflection-wf}'s fields,
    since they speak about the caller's terms.
  \end{itemize}
\end{definition}

\begin{definition}[Model-side reading of a checked condition, \texttt{Condition.WF}]
  \label{prim-cond:struct:condition-wf}
  \lean{Lean4Lean.Primitive.Condition.WF, Lean4Lean.Primitive.ConditionImpl.domain, Lean4Lean.Primitive.ConditionImpl.WF, Lean4Lean.Primitive.ReflectNatNat.WF}
  \leanok
  \uses{family:kernel-primitive-dsl, prim-cond:struct:reflection-wf, ind:trexprs}
  \srcloc{Lean4Lean/Verify/Environment/Primitive/Condition.lean}{150}

  \lead{In short} The \hl{model-side reading} of a checked condition.
  \begin{itemize}
  \item Translations $\mathtt{prop}'$, $\mathtt{dec}'$ of the predicate and decision procedure.
  \item \texttt{ConditionImpl.WF}: \texttt{Unit} for \texttt{.bool}; a \texttt{ReflectNatNat.WF}
    for \texttt{reflectNatNat} (the reflection's own \texttt{WF} plus translations of
    \texttt{asBool}, \texttt{proof}).
  \item \texttt{propT}: $\mathtt{prop}'$ applied to arguments of \texttt{ConditionImpl.domain}'s
    types is a proposition, stated applied since that is what the checks give.
  \item \texttt{vnil}: \texttt{c.vlctx = []} -- the check ran before any binder.
  \end{itemize}
\end{definition}

  \label{prim-cond:family:condition-wf-projections}
  \lean{Lean4Lean.Primitive.Condition.WF.hprop0, Lean4Lean.Primitive.Condition.WF.hdec0, Lean4Lean.Primitive.Condition.WF.propC, Lean4Lean.Primitive.Condition.WF.decC}

  \item \texttt{propC}, \texttt{decC}: their translations $\mathtt{prop}'$, $\mathtt{dec}'$ are

  \end{itemize}

\begin{definition}[Building a checked conditional: the positive specifications]
  \label{prim-cond:def:condition-wf-ite-dite}
  \lean{Lean4Lean.Primitive.ConditionImpl.WF.apply, Lean4Lean.Primitive.Condition.WF.WF_ite, Lean4Lean.Primitive.Condition.WF.WF_dite}
  \leanok
  \uses{prim-cond:struct:condition-wf, prim-cond:def:ite-dite-app}
  \srcloc{Lean4Lean/Verify/Environment/Primitive/Condition.lean}{189}

  \lead{In short} Building a checked conditional: \hl{when the decision is known}.
  \begin{itemize}
  \item \texttt{ConditionImpl.WF.apply}: the scrutinee itself for \texttt{.bool}; $\mathtt{asBool}'$
    at the arguments for a \texttt{reflectNatNat}.
  \item \texttt{WF\_ite}: \texttt{Condition.ite} at translated, well-typed pieces is exactly
    $\mathtt{iteApp}\,\alpha'\,(\mathtt{prop}'\,\mathit{args}')\,(\mathtt{dec}'\,\mathit{args}')\,t'\,e'$.
  \item \texttt{WF\_dite}: likewise for \texttt{dite}, branches read one binder deep under the
    predicate (then) and its negation (else); both quantify over an arbitrary well-formed
    \texttt{MLCtx}.
  \item \lead{Caveat} \texttt{natOnly}: nothing pins \texttt{Nat}'s universe, so a \texttt{.bool}
    condition's \texttt{ite} types only at \texttt{Nat}.
  \end{itemize}
\end{definition}

\begin{definition}[Evaluating a checked conditional]
  \label{prim-cond:def:condition-wf-eval}
  \lean{Lean4Lean.Primitive.Condition.WF.DiteEval, Lean4Lean.Primitive.Condition.WF.IteEval, Lean4Lean.Primitive.Condition.WF.DecT}
  \leanok
  \uses{prim-cond:def:condition-wf-ite-dite, prim-cond:def:ite-dite-app}
  \srcloc{Lean4Lean/Verify/Environment/Primitive/Condition.lean}{260}

  \lead{In short} What the checked conditional is worth once the \hl{decision is settled}.
  \begin{itemize}
  \item \texttt{IteEval}: if the decision at the arguments is worth boolean $b$, $\mathtt{iteApp}\,\alpha'\,(\mathtt{prop}'\,\mathit{args}')\,(\mathtt{dec}'\,\mathit{args}')\,t'\,e'$ is
    definitionally $t'$ or $e'$ accordingly.
  \item \texttt{DiteEval}: the same for \texttt{diteApp}, producing the proof its branch is
    applied to.
  \item \texttt{DecT}: $\mathtt{dec}'$ is well formed at arguments of the \texttt{impl}'s domain
    types.
  \item Split from \ref{prim-cond:def:condition-wf-ite-dite}: a caller \emph{building} a
    conditional still has an \texttt{Expr}; one that closed off probe variables does not.
  \end{itemize}
\end{definition}

\begin{definition}[Projection-free expressions, \texttt{noProj}]

\section{Transport between binder depths}

The reflection's pieces are checked at several binder depths. Nothing in \texttt{TrExprS}
identifies two readings of the same term, so this section builds the transport kit that does.

\begin{lemma}[Appending local contexts, and literals are projection-free]
  \label{prim-cond:family:vlctx-append}
  \lean{Lean4Lean.Primitive.VLCtx.toCtx_append, Lean4Lean.Primitive.VLCtx.find?_append_inv, Lean4Lean.Primitive.VLCtx.find?_append, Lean4Lean.Primitive.noProj_toConstructor}
  \leanok
  \uses{def:vt-vlctx, prim-cond:def:noproj}
  \srcloc{Lean4Lean/Verify/Environment/Primitive/Condition.lean}{438}

  \lead{In short} Appending local contexts, and \hl{literals are projection-free}.
  \begin{itemize}
  \item \texttt{toCtx\_append}: $(\Delta \mathbin{+\!\!+} \Delta').\mathtt{toCtx} =
    \Delta.\mathtt{toCtx} \mathbin{+\!\!+} \Delta'.\mathtt{toCtx}$.
  \item \texttt{find?\_append}/\texttt{find?\_append\_inv}: a lookup landing in the prefix agrees
    with the lookup in the appended context.
  \item \texttt{noProj\_toConstructor}: a literal's constructor unfolding contains no projections.
  \end{itemize}
\end{lemma}
\begin{proof}
  \leanok
  \lead{Idea} Structural recursion on the prefix for the first three; \texttt{noProj\_toConstructor}
  splits on \texttt{Nat}/\texttt{String} literals and inducts on the string's characters.
\end{proof}

\begin{lemma}[A translation survives binders appended on the right]
  \label{prim-cond:thm:trexprs-weakr}
  \lean{Lean4Lean.Primitive.TrExprS.weakR}
  \leanok
  \uses{ind:trexprs, prim-cond:def:noproj}
  \srcloc{Lean4Lean/Verify/Environment/Primitive/Condition.lean}{485}
  \thesisref{Weakening (typesys.tex, thm:weak)}

  \lead{In short} Adding context on the \hl{right} does not change a term's reading.
  \begin{itemize}
  \item \lead{Given} an ordered environment, $e$ translates to $e'$ in $\Delta$, $\Delta
    \mathbin{+\!\!+} \Delta'$ well formed, $e$ projection-free.
  \item \lead{Then} $e$ translates to the same $e'$ in $\Delta \mathbin{+\!\!+} \Delta'$.
  \end{itemize}
\end{lemma}
\begin{proof}
  \uses{prim-cond:family:vlctx-append}
  \leanok
  \lead{Idea} Induction on the \texttt{TrExprS} derivation: variables via
  \texttt{VLCtx.find?\_append}; binder cases need the typing side conditions weakened, via local
  helpers built from \texttt{VEnv.IsDefEq.weakR} and \texttt{toCtx\_append}.
\end{proof}

\begin{theorem}[A closed term is read the same way wherever it is read]
  \label{prim-cond:thm:trexprs-ofclosed}
  \lean{Lean4Lean.Primitive.TrExprS.ofClosed}
  \leanok
  \uses{ind:trexprs, prim-cond:def:noproj}
  \srcloc{Lean4Lean/Verify/Environment/Primitive/Condition.lean}{540}
  \thesisref{Weakening, parts 3--4, i.e.\ strengthening (typesys.tex, thm:weak)}

  \lead{In short} A \hl{closed term} is read the same way at any narrower or wider context.
  \begin{itemize}
  \item \lead{Given} $\mathit{env}$ well formed, $e$ translates to $e'$ in $\mathit{pre}
    \mathbin{+\!\!+} \Delta$, $e$ projection-free with no free variables and no bound variable
    escaping $\mathit{pre}$, $\mathit{pre}$ has closed types and values, $e'$ well formed there.
  \item \lead{Then} $e'$ is closed over $\mathit{pre}$ and $e$ already translates to $e'$ in
    $\mathit{pre}$ alone.
  \end{itemize}
  Converse of \ref{prim-cond:thm:trexprs-weakr}: no check has to re-run to read at the base context.
\end{theorem}
\begin{proof}
  \uses{prim-cond:family:vlctx-append, fam:strong-inversions}
  \leanok
  \lead{Idea} Induction on the \texttt{TrExprS} derivation, proving closedness of $e'$ alongside
  the narrowed translation via the \texttt{weakN\_iff} lemmas; binder cases are re-derived at the
  narrower context, not transported.
  \axfoot{sorryAx, inherited: those three weakN\_iff lemmas and OnCtx.weakN\_inv live in
  Theory/Typing/UniqueTyping.lean and are sorry-tainted on master already}
\end{proof}

\begin{lemma}[Identifying a closed piece wherever it is read]
  \label{prim-cond:family:trexprs-nil-inv}
  \lean{Lean4Lean.Primitive.TrExprS.of_nil_any, Lean4Lean.Primitive.TrExprS.of_nil_unique, Lean4Lean.Primitive.TrExprS.app2_nil_inv, Lean4Lean.Primitive.TrExprS.app1_nil_inv, Lean4Lean.Primitive.Condition.WF.prop_app2_inv, Lean4Lean.Primitive.Condition.WF.dec_app2_inv}
  \leanok
  \uses{prim-cond:def:noproj, prim-cond:struct:condition-wf}
  \srcloc{Lean4Lean/Verify/Environment/Primitive/Condition.lean}{664}

  \lead{In short} A closed translation is \hl{unique}, and so is any application built on it.
  \begin{itemize}
  \item \texttt{of\_nil\_any}: a projection-free translation at the empty context holds at every
    context.
  \item \texttt{of\_nil\_unique}: it is then the only translation that term has, anywhere.
  \item \texttt{app1\_nil\_inv}/\texttt{app2\_nil\_inv}: an application of such a closed head to one
    or two arguments translates to the head's fixed translation applied to the arguments'.
  \item \texttt{prop\_app2\_inv}/\texttt{dec\_app2\_inv}: the same specialised to a condition's
    \texttt{prop}/\texttt{dec}.
  \end{itemize}
\end{lemma}
\begin{proof}
  \uses{prim-cond:thm:trexprs-weakr}
  \leanok
  \lead{Idea} \texttt{of\_nil\_any} is \ref{prim-cond:thm:trexprs-weakr} at an empty prefix;
  \texttt{of\_nil\_unique} combines it with \texttt{TrExprS.unique'} and \texttt{noProj.isUnique}.
  The inversions strip the spine and identify the head by \texttt{of\_nil\_unique}; the two
  \texttt{app2} forms let \texttt{Bitwise.lean} and \texttt{DivMod.lean} recognise a condition
  under several binders.
\end{proof}

\section{Closed source types}

\begin{lemma}[Environment-free closed types]

  \srcloc{Lean4Lean/Verify/Environment/Primitive/Condition.lean}{728}

  \lead{In short} Two types that translate in \hl{any context}, needing nothing from the

\section{The reflection layer}

A \texttt{Reflection} packages a relation \texttt{type} between a proposition and a boolean, a
decision procedure \texttt{toDec}, and the \texttt{ite}/\texttt{dite} gadgets built on it.
\texttt{Reflection.check} establishes the unconditional part; \texttt{checkITE} and
\texttt{checkNatDITE} are the two optional checks.

\begin{theorem}[The fuel recursion shared by \texttt{Nat.div} and \texttt{Nat.mod}]

\section*{Caveats}

\label{sec:prim-arith-review}
\begin{itemize}
