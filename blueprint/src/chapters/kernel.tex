\leanok

\lead{In short} The value a constant unfolds to under $\delta$: \texttt{defnInfo} and
\texttt{thmInfo} have one, every other kind does not.
\begin{itemize}
\item \lead{Caveat} Differs from Lean's \texttt{value?} (lean4\#12973 excludes theorems); the

\begin{definition}[Monadic expression replacement, \texttt{replaceM}]
\label{def:kernel-expr-replace}
\lean{Lean.Expr.replaceM, Lean.Expr.replaceNoCacheT, Lean.Expr.ReplaceImpl.replaceUnsafe', Lean.Expr.ReplaceImpl.replaceUnsafeT}
\leanok
\lead{In short} Rewrites $e$ top-down by a partial function \texttt{f?}, recursing into subterms

\begin{itemize}
\item \texttt{replaceNoCacheT} is the structural reference; a memoizing \texttt{unsafe} version

\item \lead{Caveat} \texttt{replaceM} itself is \texttt{partial}, so it has no equation lemmas; the

\begin{definition}[Literal expansion to constructor applications]
\label{def:kernel-lit-to-ctor}
\lean{Lean.Expr.natLitToConstructor, Lean.Expr.strLitToConstructor, Lean.Literal.toConstructor, Lean.Literal.typeName}
\leanok
\lead{In short} Expands a compact literal into a constructor application, so the $\iota$ rule can
fire on it: $0 \mapsto \mathsf{Nat.zero}$, $n{+}1 \mapsto \mathsf{Nat.succ}\;n$; a string becomes
\texttt{String.ofList} applied to a \texttt{List Char} of \texttt{Char.ofNat} applications.
\texttt{typeName} names a literal's type.
\begin{itemize}
\item This is why \texttt{Nat}'s primitive status
(Definition~\ref{def:kernel-primitive-checkinductive}) is a soundness obligation: it lets $\iota$
fire on literals.
\end{itemize}
\end{definition}

\begin{definition}[Substitution-free beta reduction, \texttt{cheapBetaReduce}]
\label{def:kernel-cheap-beta}
\lean{Lean.Expr.cheapBetaReduce}
\leanok
\thesisref{axioms.tex \S2.2, the substitution-free $\beta$ rule}

\lead{In short} Beta-reduces \hl{without substitution}: $(\lambda x_1\dots x_n.b)\,a_1\dots a_m$
reduces only to $b\,a_{n+1}\dots a_m$ (no loose bound variables in $b$) or $a_i\,a_{n+1}\dots a_m$
($b$ is $x_i$); otherwise $e$ is unchanged.
\begin{itemize}
\item \lead{Caveat} The $x_i$ branch is guarded by \texttt{assert! n < i} at
\srcloc{Lean4Lean/Instantiate.lean}{24}; the invariant is not established here.
\end{itemize}
\end{definition}

\leanok

\lead{In short} How context extension $\Gamma, x{:}\alpha$ becomes an operation on a concrete

\end{definition}

\section{Universe levels}

Deciding $\ell \equiv \ell'$ and $\ell \le \ell' + n$.

\begin{definition}[Undefined level parameter detection, \texttt{getUndefParam}]
\label{def:kernel-level-getundefparam}
\lean{Lean.Level.getUndefParam, Lean.Level.forEach}
\leanok
\thesisref{axioms.tex \S2.5, the universe-scope side condition}

\lead{In short} Returns some parameter of $\ell$ absent from $\bar u$, via a short-circuiting
\texttt{forEach}.
\end{definition}

\begin{definition}[Canonical form for universe levels, \texttt{Normalize.normalize}]
\label{def:kernel-level-normalize}
\lean{Lean.Level.Normalize.normalize, Lean.Level.Normalize.normalizeAux, Lean.Level.Normalize.NormLevel, Lean.Level.Normalize.Node, Lean.Level.Normalize.VarNode, Lean.Level.Normalize.NormLevel.subsumption, Lean.Level.Normalize.NormLevel.toTree, Lean.Level.Normalize.Tree.reify, Lean.Level.normalize'}
\leanok
\thesisref{axioms.tex \S2.2, the level syntax}

\lead{In short} A canonical form for $\ell ::= u \mid 0 \mid \ell+1 \mid \max \mid \mathrm{imax}$
(after G\'eran): a map from $\mathrm{imax}$ condition sets to a \texttt{Node} of the largest
constant and offset per parameter. \texttt{subsumption} deletes dominated sublevels;
\texttt{toTree}/\texttt{Tree.reify} read a canonical \texttt{Level} back out.
\begin{itemize}
\item \lead{Status} Chapter~\ref{chap:levels} proves \texttt{normalize} complete, sorry-free.
\end{itemize}
\end{definition}

\begin{definition}[Sublevel comparison on canonical forms, \texttt{NormLevel.le}]
\label{def:kernel-level-le}
\lean{Lean.Level.Normalize.NormLevel.le, Lean.Level.Normalize.NormLevel.addable, Lean.Level.Normalize.NormLevel.feasible, Lean.Level.Normalize.NormLevel.lexChain}
\leanok
\uses{def:kernel-level-normalize}
\thesisref{axioms.tex \S2.2, $\ell \le \ell' + n$}

\lead{In short} Decides $\ell_1 \le \ell_2$ on canonical forms: every sublevel of $\ell_1$ must be
dominated by sublevels of $\ell_2$. Each entry of $\ell_2$ discharges what it can of a node of
$\ell_1$, dominated once nothing is left. \texttt{addable}/\texttt{feasible}/\texttt{lexChain}
build the least admissible chain used by reification.
\begin{itemize}
\item G\'eran's paper is cited at \srcloc{Lean4Lean/Level.lean}{27}.
\item \lead{Status} Chapter~\ref{chap:levels} proves \texttt{NormLevel.le} complete, sorry-free.
\end{itemize}
\end{definition}

\begin{definition}[Complete level equality and comparison, \texttt{isEquiv'}]
\label{def:kernel-level-isequiv}
\lean{Lean.Level.isEquiv', Lean.Level.isEquivList, Lean.Level.geq'}
\leanok
\uses{def:kernel-level-normalize, def:kernel-level-le}
\thesisref{axioms.tex \S2.2, antisymmetry gives $\ell \equiv \ell'$}

\lead{In short} $\ell_1 \equiv \ell_2$ runs core Lean's incomplete \texttt{isEquiv} first, falling
back to \hl{equality of canonical forms}; $\ell_1 \ge \ell_2$ falls back to \texttt{NormLevel.le}.
\texttt{isEquivList} lifts this pointwise over a constant's level arguments.
\begin{itemize}
\item \lead{Caveat} Accepts strictly more pairs than the standard library or the C++ kernel: it can
accept declarations they reject for level incompleteness, never the converse.
\end{itemize}
\end{definition}

\section{Pointer equality, fuel and the equivalence cache}

Two trusted axioms, a fuel budget per loop, and a cache of definitional equalities.

\begin{definition}[Pointer equality and its soundness axioms]
\label{axiom:kernel-ptr-eq}
\lean{Lean4Lean.ptrEqExpr, Lean4Lean.ptrEqExpr_eq, Lean4Lean.ptrEqConstantInfo, Lean4Lean.ptrEqConstantInfo_eq}
\leanok
\axfoot{Lean4Lean.ptrEqExpr\_eq; Lean4Lean.ptrEqConstantInfo\_eq; Classical.choice}

\lead{In short} \texttt{ptrEqExpr}/\texttt{ptrEqConstantInfo}: \texttt{opaque} Booleans from
pointer comparison; the \texttt{\_eq} \texttt{axiom}s say \texttt{true} implies actual equality.
\hl{The converse is not assumed}. \texttt{withPtrEq} cannot be used: the kernel's behaviour, not
just its speed, depends on pointer identity.
\begin{itemize}
\item \lead{Caveat} At \srcloc{Lean4Lean/TypeChecker.lean}{739} \texttt{ptrEqConstantInfo} gates a
different algorithm (congruence before unfolding), not merely a shortcut. Used at
\srcloc{Lean4Lean/Verify/EquivManager.lean}{264} and
\srcloc{Lean4Lean/Verify/TypeChecker/IsDefEq.lean}{384}.
\end{itemize}
\end{definition}

\begin{definition}[Union-find cache of definitional equalities, \texttt{EquivManager}]
\label{def:kernel-equiv-manager}
\lean{Lean4Lean.EquivManager, Lean4Lean.EquivManager.isEquiv, Lean4Lean.EquivManager.addEquiv, Lean4Lean.EquivManager.find, Lean4Lean.EquivManager.merge, Lean4Lean.EquivManager.toNode}
\leanok
\uses{axiom:kernel-ptr-eq}
\lead{In short} A \texttt{Batteries.UnionFind} over expressions recording pairs already found
definitionally equal. \texttt{isEquiv}: short-circuits on pointer equality, a hash mismatch, or a
shared root; else recurses congruentially and merges roots on success. \texttt{addEquiv} records a
successful \texttt{isDefEq} verdict.
\begin{itemize}
\item \lead{Caveat} The algorithmic relation is not transitive, so the closure can claim
equalities the algorithm alone would not decide; \texttt{isEquiv} is not a conservative
optimization.
\end{itemize}
\end{definition}

\begin{definition}[Validation of the prelude's \texttt{Eq}, \texttt{checkEqType}]
\label{def:kernel-check-eq-type}
\contrib{mixed} \srcloc{Lean4Lean/Quot.lean}{20}
\lean{Lean4Lean.checkEqType}
\leanok
\uses{family:kernel-env-checks, def:kernel-exprbuildt, def:kernel-monad-localnamegen}
\thesisref{axioms.tex \S2.6.2, $\mathsf{eq}_a := \mu\,t{:}\alpha\to\mathsf{Prop}.\,(\mathsf{refl}:t\;a)$}
\axfoot{propext; Classical.choice; Quot.sound (no sorryAx)}

\lead{In short} Checks, before quotient initialization, that \texttt{Eq} has exactly the
prelude's shape: one universe parameter $u$, one constructor; type
$\forall\,\{\alpha:\mathsf{Sort}\;u\},\alpha\to\alpha\to\mathsf{Prop}$; constructor's type
$\forall\,\{\alpha:\mathsf{Sort}\;u\}(a:\alpha),\mathsf{Eq}\;\alpha\;a\;a$ (up to binder names).
\begin{itemize}
\item \textbf{On \texttt{iota}} (\srcloc{Lean4Lean/Quot.lean}{24}): also not \texttt{unsafe}
(reject-more: \texttt{Quot.lift}'s type mentions \texttt{Eq}, so an unsafe \texttt{Eq} would leave
a safe constant depending on an unsafe one), discharging \texttt{addQuot.WF}/\texttt{addDecl.WF}'s
\texttt{Environment.EqSafe} hypothesis.
\item \lead{Caveat} \texttt{addQuot} returns early once \texttt{quotInit} is set, and
\texttt{finalizeImport} sets it for any non-empty imports, so neither check runs on an imported
environment.
\end{itemize}
\end{definition}

\begin{definition}[Quotient initialization, \texttt{addQuot}]
\label{def:kernel-add-quot}
\lean{Lean4Lean.Environment.addQuot}
\leanok
\uses{def:kernel-check-eq-type, family:kernel-env-checks, def:kernel-exprbuildt, def:kernel-monad-localnamegen}
\thesisref{axioms.tex \S2.7, the quotient constants}

\lead{In short} Unchanged once \texttt{quotInit} is set; otherwise runs \texttt{checkEqType},
checks the four names are free, and adds \texttt{Quot}, \texttt{Quot.mk}, \texttt{Quot.lift},
\texttt{Quot.ind} as \texttt{quotInfo} with types built as telescopes. \texttt{Quot.sound} stays
an ordinary prelude axiom, as in the thesis, and is not added here.
\end{definition}

\begin{definition}[Quotient computation rule, \texttt{quotReduceRec}]
\label{def:kernel-quot-reduce-rec}
\lean{Lean4Lean.quotReduceRec}
\leanok
\thesisref{axioms.tex \S2.7, $\mathrm{lift}_R\;\beta\;f\;h\;(\mathrm{mk}_R\;a) \rightsquigarrow
f\;a$}

\lead{In short} Reduces $\mathrm{lift}\;\alpha\;r\;\beta\;f\;h\;(\mathrm{mk}\;r\;a)\;\bar c
\rightsquigarrow f\;a\;\bar c$ and $\mathrm{ind}\;\dots\;p\;(\mathrm{mk}\;r\;a)\;\bar c
\rightsquigarrow p\;a\;\bar c$.
\begin{itemize}
\item Normalizes the major at its fixed position (5/4), requires a \texttt{Quot.mk} of arity 3,
applies the function argument (index 3) to the last argument, re-applies trailing arguments.
\item \lead{Caveat} Unlike the inductive $\iota$ rule, nothing checks saturation: a partially
applied \texttt{Quot.lift} simply fails to match.
\end{itemize}
\end{definition}

\lean{Lean4Lean.getFirstCtor, Lean4Lean.mkNullaryCtor}
\leanok
\lead{In short} \texttt{getFirstCtor}: the head of an inductive's constructor list.
\texttt{mkNullaryCtor type nparams}: decomposes $I\;\bar a$ and returns $I$'s first constructor at

\end{definition}

\begin{definition}[K-like reduction, \texttt{toCtorWhenK}]
\label{def:kernel-to-ctor-when-k}
\lean{Lean4Lean.toCtorWhenK}
\leanok
\uses{def:kernel-get-first-ctor}
\thesisref{axioms.tex \S2.6.4, the K-like reduction rule}

\lead{In short} For a recursor with \texttt{rval.k}, replaces a major premise $e$ by the nullary
constructor application of its own type, when the types are definitionally equal (pinning indices)
and no index carries a metavariable; otherwise $e$ is unchanged.
\begin{itemize}
\item \lead{Caveat} Proof irrelevance then $\iota$; \texttt{reduceRecursor.WF}
(Definition~\ref{def:kernel-reduce-recursor}) stays open, as this step has no \texttt{IsDefEq}
counterpart.
\end{itemize}
\end{definition}

\begin{definition}[Structure-eta expansion of a major premise, \texttt{toCtorWhenStruct}]
\label{def:kernel-to-ctor-when-struct}
\lean{Lean4Lean.toCtorWhenStruct, Lean4Lean.expandEtaStruct}
\leanok
\uses{def:kernel-get-first-ctor, def:kernel-is-nonrec-structure}
\lead{In short} When the major's inductive is a non-recursive structure, not already a constructor
application, with universe \texttt{isNeverZero}, replaces it by $\mathsf{S.mk}\;\bar
a\;(e.0)\dots(e.(n{-}1))$ (defeq by structure $\eta$), so $\iota$ can fire. No thesis counterpart.
\begin{itemize}
\item \lead{Caveat} Tests \texttt{isNeverZero}, where the C++ kernel tests
\texttt{!isAlwaysZero}; this narrows the class of projections any \texttt{TrProj} result can
cover.
\end{itemize}
\end{definition}

\begin{definition}[Recursor rule lookup, \texttt{getRecRuleFor}]
\label{def:kernel-get-rec-rule-for}
\lean{Lean4Lean.getRecRuleFor}
\leanok
\thesisref{axioms.tex \S2.6.4, selecting $e_c$}

\lead{In short} Given a \texttt{RecursorVal} and a major premise, returns the
\texttt{RecursorRule} whose \texttt{ctor} matches the major's head constant, or \texttt{none} if
the head is not a constant or no rule matches.
\end{definition}

\begin{definition}[The pure $\iota$ step, \texttt{inductiveReduceRecCore}]
\label{def:kernel-iota-core}
\contrib{trproj} \srcloc{Lean4Lean/Inductive/Reduce.lean}{75}
\lean{Lean4Lean.inductiveReduceRecCore}
\leanok
\uses{def:kernel-get-rec-rule-for}
\thesisref{axioms.tex \S2.6.4, $\mathrm{rec}_P\;C\;e\;p[b]\;(c\;b) \rightsquigarrow e_c\;b\;v$}
\axfoot{propext; Quot.sound (no sorryAx)}

\lead{In short} A pure, monad-free \hl{\texttt{Option Expr}}-valued $\iota$ step on a major
premise already in constructor form.
\begin{itemize}
\item Instantiates the matched rule's \texttt{rhs} at \texttt{ls}, applies it to \texttt{recArgs}
up to \texttt{rval.getFirstIndexIdx}, then to the major's last \texttt{rule.nfields} arguments
(its fields), then to arguments after \texttt{rval.getMajorIdx}; \texttt{none} on no match, too
few arguments, or a level-count mismatch. Extracted verbatim from master, recomputing
\texttt{rval.getMajorIdx} (behaviour-preserving): lets

\section{Type inference}

The algorithmic judgment $\Gamma \Vdash e:\alpha$, telescope by telescope.

\begin{definition}[Typing of constants and free variables]
\label{def:kernel-infer-constant}

\leanok

With \texttt{inferOnly := false}, also rejects an unsafe constant from non-unsafe code, a

\end{definition}

\begin{definition}[Typing of binders and applications]
\label{def:kernel-infer-binders}

\lead{In short} Telescope-at-a-time: lambda gives the pi over collected binders' body type; pi
gives $\mathcal{U}_{\mathrm{imax}(\ell_1,\dots,\ell_n,\ell)}$ via \texttt{mkLevelIMax'};
\texttt{let} checks the value against the ascribed type (\texttt{inferOnly} false) and binds it;
\texttt{inferApp} walks the spine once, instantiating pi bodies lazily.
\begin{itemize}
\item \lead{Caveat} \texttt{inferApp} is the \texttt{inferOnly := true} fast path, correct only on
well-typed terms; recorded at \srcloc{Lean4Lean/TypeChecker.lean}{172}.
\end{itemize}
\end{definition}

\begin{definition}[Typing of structure projections, \texttt{inferProj}]
\label{def:kernel-infer-proj}
\lean{Lean4Lean.TypeChecker.Inner.inferProj}
\leanok
\uses{def:kernel-whnf}
\lead{In short} Type of \texttt{.proj S i e}: normalize $e$'s type to $I\;\bar a$, require $I =
\texttt{S}$, single-constructor, $|\bar a| = \texttt{numParams}+\texttt{numIndices}$.
\begin{itemize}
\item Instantiates the constructor's type at the structure's levels and parameters, walks $i$
binders (each dependent field via \texttt{.proj S j e}), returns the next binder's domain. If the
universe is not \texttt{isNeverZero}, every dependent field's domain and the result must also be a
proposition (\texttt{isProp}, Definition~\ref{def:kernel-proof-irrel}). No thesis counterpart.
\item \lead{Caveat} Tests \texttt{isNeverZero}, not the C++ kernel's \texttt{!isAlwaysZero},
keeping \texttt{proj} no stronger than the generated recursor; bounds \texttt{inferProj.WF} and
the \texttt{TrProj} results.
\end{itemize}
\end{definition}

\begin{definition}[The algorithmic typing judgment, \texttt{inferType'}]
\label{def:kernel-infer-type}
\lean{Lean4Lean.TypeChecker.Inner.inferType', Lean4Lean.TypeChecker.Inner.inferType}
\leanok
\uses{def:kernel-infer-constant, def:kernel-infer-binders, def:kernel-infer-proj}
\thesisref{typesys.tex \S3, failure of subject reduction}

\lead{In short} The main recursion: loose bound variables rejected outright, results memoized per
\texttt{inferOnly}. False: the full judgment, checking level scopes, the literal types
\texttt{Nat}/\texttt{Char.ofNat}/\texttt{String.ofList}, and, per application, defeq of the
argument's type with the domain (Definition~\ref{def:kernel-isdefeq-core}). True: assumes
well-typedness, takes the \texttt{inferApp} fast path. The \texttt{eagerReduce} marker
(\srcloc{Lean4Lean/TypeChecker.lean}{299}) switches strategy on a
syntactic test, mirroring core Lean.
\end{definition}

and a valid application of its own type. \texttt{declareConstructors}: adds each
\texttt{ctorInfo}, field count $=$ arity $-$ \texttt{numParams}.
\end{definition}

\begin{definition}[Large elimination and K-likeness]

\thesisref{axioms.tex \S2.6.2, the $\mathsf{LE}$ judgments}

\lead{In short} \texttt{isLargeEliminator}: true when the result universe is provably nonzero, or

\end{definition}

\begin{definition}[Motives and minor premises, \texttt{mkRecInfos}]
\label{def:kernel-mk-rec-infos}
\lean{Lean4Lean.AddInductive.mkRecInfos, Lean4Lean.AddInductive.mkRecInfos.loopArgs1, Lean4Lean.AddInductive.mkRecInfos.loopInd1, Lean4Lean.AddInductive.mkRecInfos.loopCtorArgs, Lean4Lean.AddInductive.mkRecInfos.loopUArgs, Lean4Lean.AddInductive.mkRecInfos.loopU, Lean4Lean.AddInductive.mkRecInfos.loopCtors, Lean4Lean.AddInductive.mkRecInfos.loopInd2}

$C:\forall \bar i,\ I\;\bar a\;\bar i \to \mathcal{U}_u$ at the elimination level.
Per constructor, the minor premise $\varepsilon_c = \forall b{::}\beta,\forall v{::}\delta,\
C\;p[b]\;(c\;b)$ with $\delta_i = \forall x{::}\xi_i,\ C\;\pi_i[b,x]\;(u_i\;x)$ for each recursive
argument (\texttt{isRecArg}).
\end{definition}

\begin{definition}[Generation of the $\iota$ rules, \texttt{mkRecRules}]
\label{def:kernel-mk-rec-rules}
\lean{Lean4Lean.AddInductive.mkRecRules, Lean4Lean.AddInductive.getRecLevels, Lean4Lean.AddInductive.getRecLevelParams}

\lead{In short} Per constructor, builds the \texttt{RecursorRule} with \texttt{nfields} the
non-parameter arity and right-hand side $\lambda \bar a\,\bar C\,\bar e\,b{::}\beta.\ e_c\;b\;v$,
where $v_i = \lambda x{::}\xi_i.\ \mathrm{rec}_{P_j}\;\bar a\;\bar C\;\bar e\;\pi_i[b,x]\;(u_i\;x)$.

\texttt{inductiveReduceRecCore} (Definition~\ref{def:kernel-iota-core}) later consumes.
\end{definition}

\section*{Caveats}

\begin{itemize}
\item \texttt{addDecl.WF}'s \texttt{inductDecl} case is \texttt{sorry}
(Definition~\ref{def:kernel-add-inductive}); inductive-block guarantees rest on that gap.
\item \texttt{reduceRecursor.WF} is open: K-like and structure-$\eta$ have no \texttt{IsDefEq}
counterpart; only the pure $\iota$ core is verified.
\item \texttt{ElimNestedInductive} is fully unverified, with a totality hole in
\texttt{restoreCtorName}.
\item \texttt{etaExpand} has no call site, so \texttt{inductiveReduceRecCore.WF}'s saturation
hypothesis is unenforced.
\item \texttt{checkEqType}'s shape and safety checks skip an imported environment.
\item Projection and structure-$\eta$ test \texttt{isNeverZero}, not \texttt{!isAlwaysZero};
\texttt{divergences.md} documents fourteen deliberate divergences from the C++ kernel in total.
\item The kernel's core operations are mutually recursive; \texttt{\textbackslash uses} records one direction
only, since the dependency graph must be acyclic.
\end{itemize}
