% Macros and settings shared by the pdf (print.tex) and the web (web.tex)
% versions of the lean4lean blueprint.
%
% Format-specific overrides live in macros/print.tex and macros/web.tex,
% both of which are read *after* this file.

% ---------------------------------------------------------------------------
% Theorem-like environments.
%
% The five environments definition/lemma/proposition/theorem/corollary are the
% dependency-graph node types recognised by plastexdepgraph.  This blueprint
% additionally uses `example' (for the declarations coming from test/example
% files); it is added to the graph through the `thms=' package option in
% web.tex.  `remark' is deliberately NOT a graph node: it is plain commentary.
% All of them share the `theorem' counter, which is never reset.
% ---------------------------------------------------------------------------
% Every node environment is upright (definition style): node bodies are schematic -- lists,
% run-in labels, badges -- which read badly in the italics of the plain style.
\theoremstyle{definition}
\newtheorem{theorem}{Theorem}
\newtheorem{proposition}[theorem]{Proposition}
\newtheorem{lemma}[theorem]{Lemma}
\newtheorem{corollary}[theorem]{Corollary}

\theoremstyle{definition}
\newtheorem{definition}[theorem]{Definition}
\newtheorem{example}[theorem]{Example}

\theoremstyle{remark}
\newtheorem{remark}[theorem]{Remark}

\theoremstyle{plain}

% ---------------------------------------------------------------------------
% Blueprint-specific annotation macros.
%
% Usage rules (see the authoring conventions):
%   \contrib{iota|trproj|mixed}  first thing in the body of every node whose
%                                attribution is not `master'.
%   \srcloc{path}{line}          source pointer; underscores are written raw
%                                inside this macro (no Lean4Lean source path
%                                contains one).  Overridden in macros/web.tex
%                                to link to GitHub.
%   \thesisref{...}              thesis correspondence line.
%   \axfoot{...}                 axiom footprint line.
%   \reviewnote{...}             a critical remark; no fixes.
% ---------------------------------------------------------------------------
% Note: the trailing \space (a plain, breakable interword space) matters:
% a control space \ is not a legal line break, which produced overfull
% boxes in the pdf when a badge was followed by a long \srcloc path.
% ---------------------------------------------------------------------------
% Schematic-style helpers (see blueprint/STYLE.md).
% ---------------------------------------------------------------------------
% Run-in label opening a line of a node: \lead{In short}, \lead{Given}, \lead{Then}, ...
\newcommand{\lead}[1]{\textbf{#1.}}
% The one phrase of a node a skimming reader must not miss (at most one per node).
\definecolor{bphl}{HTML}{9A3B00}
\newcommand{\hl}[1]{\textcolor{bphl}{\textbf{#1}}}
% Status badges.
\definecolor{bpproved}{HTML}{1B7F3B}
\definecolor{bpsorry}{HTML}{B00020}
\definecolor{bptainted}{HTML}{B35900}
\definecolor{bpcontrib}{HTML}{0B6E99}
\definecolor{bpmeta}{HTML}{555555}
\newcommand{\stProved}{\textcolor{bpproved}{\textbf{[PROVED]}}}         % proved, no sorry in its closure
\newcommand{\stSorry}{\textcolor{bpsorry}{\textbf{[SORRY]}}}            % the Lean proof contains a literal sorry
\newcommand{\stTainted}{\textcolor{bptainted}{\textbf{[USES SORRY]}}}   % proved, but depends on a sorry elsewhere
\newcommand{\stExperimental}{\textcolor{bpmeta}{\textbf{[EXPERIMENTAL]}}} % research module, not on the verified path

% Attribution badge: first thing in the body of every node not inherited unchanged from upstream.
%   iota   = added by the iota contribution (rules for recursor reduction on inductive blocks)
%   trproj = added by the projection contribution (TrProj and structure projections)
%   mixed  = an upstream declaration modified by a contribution
% The trailing \space is a breakable interword space (a control space is not a legal break).
\newcommand{\contrib}[1]{\textcolor{bpcontrib}{\textbf{[#1]}}\space}

% Source pointer, pinned to the documented commit (web: a link to the GitHub blob).
\newcommand{\srcloc}[2]{{\small\textcolor{bpmeta}{\texttt{#1:#2}}}}

% Correspondence with Carneiro's thesis.
\newcommand{\thesisref}[1]{{\small\textcolor{bpmeta}{\textbf{Thesis.} #1}}}

% Axiom footprint of a node that depends on a sorry.
\newcommand{\axfoot}[1]{\par\noindent{\small\stTainted{} \textcolor{bpmeta}{\texttt{#1}}}\par}
